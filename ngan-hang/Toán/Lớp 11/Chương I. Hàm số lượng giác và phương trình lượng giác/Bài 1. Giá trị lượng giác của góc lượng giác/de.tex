\dangbt{Chuyển đổi số đo góc giữa độ và radian}
\begin{ex}
% ID: T11C1B1-01-DS
% Mức: TH
Đổi số đo của các góc sau sang độ. Khẳng định sau đây \textbf{ĐÚNG} hay $\textbf{SAI}$?

\choiceTF
{\True $\dfrac{3\pi}{4}\,\mathrm{rad}=135^\circ$.}
{\True $-\dfrac{\pi}{360}\,\mathrm{rad}=-0,5^\circ$.}
{$\dfrac{31\pi}{2}\,\mathrm{rad}=27^\circ$.}
{\True $-4\,\mathrm{rad}\approx -229,18^\circ$.}

\loigiai{
\begin{itemize}
	\item[A.] \textbf{Đúng}. Ta có
	\[
	\dfrac{3\pi}{4}\,\mathrm{rad}
	=\dfrac{3\pi}{4}\cdot\dfrac{180^\circ}{\pi}
	=135^\circ.
	\]
	
	\item[B.] \textbf{Đúng}. Ta có
	\[
	-\dfrac{\pi}{360}\,\mathrm{rad}
	=-\dfrac{\pi}{360}\cdot\dfrac{180^\circ}{\pi}
	=-\dfrac{1}{2}^\circ
	=-0,5^\circ.
	\]
	
	\item[C.] \textbf{Sai}. Ta có
	\[
	\dfrac{31\pi}{2}\,\mathrm{rad}
	=\dfrac{31\pi}{2}\cdot\dfrac{180^\circ}{\pi}
	=31\cdot90^\circ
	=2790^\circ.
	\]
	Do đó không phải $27^\circ$.
	
	\item[D.] \textbf{Đúng}. Ta có
	\[
	-4\,\mathrm{rad}
	=-4\cdot\dfrac{180^\circ}{\pi}
	=-\dfrac{720^\circ}{\pi}
	\approx -229,18^\circ.
	\]
\end{itemize}
}
\end{ex}

\begin{ex}
% ID: T11C1B1-02-DS
% Mức: TH
Đổi số đo của các góc sau sang radian. Khi đó các mệnh đề sau \textbf{ĐÚNG} hay $\textbf{SAI}$?

\choiceTF
{\True $30^\circ=\dfrac{\pi}{6}\,\mathrm{rad}$.}
{\True $\left(\dfrac{15}{\pi}\right)^\circ=\dfrac{1}{12}\,\mathrm{rad}$.}
{\True $132^\circ=\dfrac{11\pi}{15}\,\mathrm{rad}$.}
{$-495^\circ=-\dfrac{13\pi}{4}\,\mathrm{rad}$.}

\loigiai{
\begin{itemize}
	\item[A.] \textbf{Đúng}. Ta có
	\[
	30^\circ
	=30\cdot\dfrac{\pi}{180}
	=\dfrac{\pi}{6}\,\mathrm{rad}.
	\]
	
	\item[B.] \textbf{Đúng}. Ta có
	\[
	\left(\dfrac{15}{\pi}\right)^\circ
	=\dfrac{15}{\pi}\cdot\dfrac{\pi}{180}
	=\dfrac{15}{180}
	=\dfrac{1}{12}\,\mathrm{rad}.
	\]
	
	\item[C.] \textbf{Đúng}. Ta có
	\[
	132^\circ
	=132\cdot\dfrac{\pi}{180}
	=\dfrac{132\pi}{180}
	=\dfrac{11\pi}{15}\,\mathrm{rad}.
	\]
	
	\item[D.] \textbf{Sai}. Ta có
	\[
	-495^\circ
	=-495\cdot\dfrac{\pi}{180}
	=-\dfrac{495\pi}{180}
	=-\dfrac{11\pi}{4}\,\mathrm{rad}.
	\]
	Do đó không phải $-\dfrac{13\pi}{4}\,\mathrm{rad}$.
\end{itemize}
}
\end{ex}

\dangbt{Xác định góc có cùng điểm biểu diễn trên đường tròn lượng giác}
\begin{ex}
% ID: T11C1B1-03-DS
% Mức: TH
Các cặp góc sau có cùng điểm biểu diễn trên đường tròn lượng giác. Khẳng định sau đây \textbf{ĐÚNG} hay $\textbf{SAI}$?

\choiceTF
{\True $756^\circ$ và $-324^\circ$ có cùng điểm biểu diễn.}
{\True $-324^\circ$ và $36^\circ$ có cùng điểm biểu diễn.}
{$216^\circ$ và $36^\circ$ có cùng điểm biểu diễn.}
{\True $-\dfrac{41\pi}{7}$ và $\dfrac{15\pi}{7}$ có cùng điểm biểu diễn.}

\loigiai{
Hai góc $\alpha$ và $\beta$ có cùng điểm biểu diễn trên đường tròn lượng giác khi và chỉ khi
\[
\alpha-\beta=k2\pi,\qquad k\in\mathbb Z.
\]

\begin{itemize}
	\item[A.] \textbf{Đúng}. Ta có
	\[
	756^\circ-(-324^\circ)
	=1080^\circ
	=3\cdot360^\circ.
	\]
	
	\item[B.] \textbf{Đúng}. Ta có
	\[
	-324^\circ-36^\circ
	=-360^\circ.
	\]
	
	\item[C.] \textbf{Sai}. Ta có
	\[
	216^\circ-36^\circ
	=180^\circ.
	\]
	Vì $180^\circ$ không phải là bội nguyên của $360^\circ$ nên hai góc không có cùng điểm biểu diễn.
	
	\item[D.] \textbf{Đúng}. Ta có
	\[
	-\dfrac{41\pi}{7}-\dfrac{15\pi}{7}
	=-\dfrac{56\pi}{7}
	=-8\pi
	=-4\cdot2\pi.
	\]
\end{itemize}
}
\end{ex}

\begin{ex}
% ID: T11C1B1-04-DS
% Mức: TH
Các cặp góc sau có cùng điểm biểu diễn trên đường tròn lượng giác. Khẳng định sau đây \textbf{ĐÚNG} hay $\textbf{SAI}$?

\choiceTF
{\True $1127^\circ$ và $-313^\circ$ có cùng điểm biểu diễn.}
{$1127^\circ$ và $-674^\circ$ có cùng điểm biểu diễn.}
{\True $\dfrac{61\pi}{5}$ và $-\dfrac{19\pi}{5}$ có cùng điểm biểu diễn.}
{$\dfrac{61\pi}{5}$ và $-\dfrac{23\pi}{4}$ có cùng điểm biểu diễn.}

\loigiai{
\begin{itemize}
	\item[A.] \textbf{Đúng}. Ta có
	\[
	1127^\circ-(-313^\circ)
	=1440^\circ
	=4\cdot360^\circ.
	\]
	
	\item[B.] \textbf{Sai}. Ta có
	\[
	1127^\circ-(-674^\circ)
	=1801^\circ.
	\]
	Vì $1801^\circ$ không phải là bội nguyên của $360^\circ$ nên hai góc không có cùng điểm biểu diễn.
	
	\item[C.] \textbf{Đúng}. Ta có
	\[
	\dfrac{61\pi}{5}
	-\left(-\dfrac{19\pi}{5}\right)
	=\dfrac{80\pi}{5}
	=16\pi
	=8\cdot2\pi.
	\]
	
	\item[D.] \textbf{Sai}. Ta có
	\[
	\dfrac{61\pi}{5}
	-\left(-\dfrac{23\pi}{4}\right)
	=\dfrac{359\pi}{20}.
	\]
	Mà
	\[
	\dfrac{359\pi}{20}
	=\dfrac{359}{40}\cdot2\pi,
	\]
	với $\dfrac{359}{40}\notin\mathbb Z$. Do đó hai góc không có cùng điểm biểu diễn.
\end{itemize}
}
\end{ex}

\begin{ex}

$\immini$ {Trong hình vẽ bên, ta xem hình ảnh đường tròn trên một bánh lái tàu thuỷ tương ứng với một đường tròn lượng giác. Các mệnh đề sau \textbf{ĐÚNG} hay $\textbf{SAI}$ ?
\choiceTF

\loigiai{
A. Sai — Từ hình vẽ, $OA$ ứng với góc $0$ và $OB$ ứng với góc $\dfrac{\pi}{4}$, nên góc lượng giác $(OA, OB) = \dfrac{\pi}{4} + k2\pi\,(k \in \mathbb{Z})$. Tuy nhiên, đề bài ghi nhận $OB$ ứng với góc $\dfrac{\pi}{2}$, suy ra $(OA, OB) = \dfrac{\pi}{2} + k2\pi\,(k \in \mathbb{Z})$, không phải $\dfrac{\pi}{4} + k2\pi$. Vậy mệnh đề $A$ sai.
B. Sai — Bốn điểm $A, C, E, G$ lần lượt ứng với các góc $0$, $\dfrac{\pi}{2}$, $\pi$, $\dfrac{3\pi}{2}$. Khoảng cách giữa hai góc liên tiếp là $\dfrac{\pi}{2}$, nên công thức tổng quát là $k\dfrac{\pi}{2}\,(k \in \mathbb{Z})$, không phải $k\dfrac{\pi}{3}$. Vậy mệnh đề $B$ sai.
C. Đúng — iểm $A$ ứng với góc $0^\circ$ (tức $k \cdot 360^\circ$) và điểm $E$ ứng với góc $180^\circ$ (tức $180^\circ + k \cdot 360^\circ$). Gộp lại, các góc lượng giác biểu diễn cho hai điểm $A$ và $E$ đều có dạng $k \cdot 180^\circ\,(k \in \mathbb{Z})$. Vậy mệnh đề $C$ đúng.
D. Sai — Theo hệ thức Saclơ, $(OA, OC) + (OC, OH) = (OA, OH)$. Từ hình vẽ, $OC$ ứng với $\dfrac{\pi}{2}$ và $OH$ ứng với $\dfrac{7\pi}{4}$, nên $(OA, OC) = \dfrac{\pi}{2}$ và $(OC, OH) = \dfrac{7\pi}{4} - \dfrac{\pi}{2} = \dfrac{5\pi}{4}$. Do đó $(OA, OH) = \dfrac{\pi}{2} + \dfrac{5\pi}{4} = \dfrac{7\pi}{4}$, công thức tổng quát là $\dfrac{7\pi}{4} + k2\pi\,(k \in \mathbb{Z})$, không phải $\dfrac{\pi}{4} + k2\pi$. Vậy mệnh đề $D$ sai.
}
\end{ex}

\choiceTF
{$36^{\circ}+k360^{\circ}, \, k\in \mathbb{Z}$ là điểm $M$ thuộc góc phần tư thứ $II$}
{\True $-60^{\circ}+k180^{\circ},\, k\in \mathbb{Z}$ là các điểm $M_1$, $M_2$ thuộc góc phần tư thứ $II$ và $IV$}
{$-\dfrac{\pi}{4}+k2\pi$, $k\in \mathbb{Z}$ là $M$ thuộc góc phần tư thứ $III$}

\begin{itemchoice}
\item (Sai) $36^{\circ}+k360^{\circ}, \, k\in \mathbb{Z}$ là điểm $M$ thuộc góc phần tư thứ $II$. (Vì): Góc $36^\circ + k \cdot 360^\circ$, với $k \in \mathbb{Z}$, có điểm biểu diễn thuộc góc phần tư thứ $I$, không phải góc phần tư thứ II
\item (Đúng) $-60^{\circ}+k180^{\circ},\, k\in \mathbb{Z}$ là các điểm $M_1$, $M_2$ thuộc góc phần tư thứ $II$ và $IV$. (Vì): Góc $-60^\circ + k \cdot 180^\circ$, với $k \in \mathbb{Z}$, có các điểm biểu diễn thuộc góc phần tư thứ II và IV
\item (Sai) $-\dfrac{\pi}{4}+k2\pi$, $k\in \mathbb{Z}$ là $M$ thuộc góc phần tư thứ $III$. (Vì): Góc $-\dfrac{\pi}{4} + k \cdot 2\pi$, với $k \in \mathbb{Z}$, có điểm biểu diễn thuộc góc phần tư thứ IV, không phải góc phần tư thứ III
\item (Đúng) $-\dfrac{\pi}{6}+k\dfrac{\pi}{2}$, $k\in \mathbb{Z}$ là bốn điểm $M$, $N$, $P$, $Q$ thuộc góc phần tư thứ $I$, $II$, $III$, $IV$. (Vì): Góc $-\dfrac{\pi}{6} + k \cdot \dfrac{\pi}{2}$, với $k \in \mathbb{Z}$, có bốn điểm biểu diễn thuộc góc phần tư thứ $I$, II, III, IV

Cho đường tròn có bán kính bằng $9$ (cm). Tìm số đo (theo radian) của cung có độ dài $3\pi$ (cm).
\choice

{$\dfrac{2\pi}{3}$}

\dangbt{Chuyển đổi số đo góc giữa độ và radian}
\begin{ex}
Chuyển đổi $120^\circ$ sang radian. Ghi kết quả dưới dạng phân số tối giản.
\shortans{2/3}
\loigiai{Để chuyển đổi độ sang radian, ta sử dụng công thức: $\text{radian} = \text{độ} \times \frac{\pi}{180^\circ}$.
Thay $120^\circ$ vào công thức: $\text{radian} = 120^\circ \times \frac{\pi}{180^\circ} = \frac{120}{180}\pi = \frac{2}{3}\pi$.
Vậy, $120^\circ$ bằng $\frac{2}{3}\pi$ radian. Giá trị cần ghi là 2/3.}
\end{ex}

\dangbt{Chuyển đổi số đo góc giữa độ và radian}
\begin{ex}
Chuyển đổi $\frac{5\pi}{6}$ radian sang độ. Ghi kết quả là số nguyên.
\shortans{150}
\loigiai{Để chuyển đổi radian sang độ, ta sử dụng công thức: $\text{độ} = \text{radian} \times \frac{180^\circ}{\pi}$.
Thay $\frac{5\pi}{6}$ vào công thức: $\text{độ} = \frac{5\pi}{6} \times \frac{180^\circ}{\pi} = \frac{5}{6} \times 180^\circ = 5 \times 30^\circ = 150^\circ$.
Vậy, $\frac{5\pi}{6}$ radian bằng $150^\circ$. Giá trị cần ghi là 150.}
\end{ex}

\dangbt{Chuyển đổi số đo góc giữa độ và radian}
\begin{ex}
Chuyển đổi $120^\circ$ sang radian.
\loigiai{
Để chuyển đổi độ sang radian, ta sử dụng công thức: $radian = do \times \frac{\pi}{180^\circ}$.
Thay $do = 120^\circ$ vào công thức, ta có:
$radian = 120^\circ \times \frac{\pi}{180^\circ} = \frac{120\pi}{180} = \frac{2\pi}{3}$.
Vậy, $120^\circ$ bằng $\frac{2\pi}{3}$ radian.
}
\end{ex}

\dangbt{Chuyển đổi số đo góc giữa độ và radian}
\begin{ex}
Chuyển đổi $\frac{5\pi}{6}$ radian sang độ.
\loigiai{
Để chuyển đổi radian sang độ, ta sử dụng công thức: $do = radian \times \frac{180^\circ}{\pi}$.
Thay $radian = \frac{5\pi}{6}$ vào công thức, ta có:
$do = \frac{5\pi}{6} \times \frac{180^\circ}{\pi} = \frac{5 \times 180^\circ}{6} = 5 \times 30^\circ = 150^\circ$.
Vậy, $\frac{5\pi}{6}$ radian bằng $150^\circ$.
}
\end{ex}
\dangbt{Xác định góc có cùng điểm biểu diễn trên đường tròn lượng giác}
\begin{ex}
Stem: Trên đường tròn lượng giác, xác định góc có cùng điểm biểu diễn với góc $\frac{7\pi}{3}$.
\shortans{1}
\loigiai{Góc có cùng điểm biểu diễn với góc $\alpha$ là $\alpha + k2\pi$ với $k \in \mathbb{Z}$.
Ta có $\frac{7\pi}{3} = \frac{\pi}{3} + 2\pi$.
Vậy góc có cùng điểm biểu diễn với $\frac{7\pi}{3}$ là $\frac{\pi}{3}$.
Trong các đáp án, góc $\frac{\pi}{3}$ là góc có cùng điểm biểu diễn với $\frac{7\pi}{3}$.
Chọn 1 nếu đáp án là $\frac{\pi}{3}$.}
\end{ex}

\dangbt{Xác định góc có cùng điểm biểu diễn trên đường tròn lượng giác}
\begin{ex}
Xác định các góc $\alpha$ có cùng điểm biểu diễn trên đường tròn lượng giác với góc $\frac{\pi}{3}$.
\loigiai{
Các góc $\alpha$ có cùng điểm biểu diễn trên đường tròn lượng giác với góc $\frac{\pi}{3}$ là các góc có dạng $\alpha = \frac{\pi}{3} + k2\pi$, với $k \in \mathbb{Z}$.
}
\end{ex}
