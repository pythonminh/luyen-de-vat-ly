\begin{ex}
% ID: T10C1B1-01-DS
% Mức: NB
Xét các câu sau đây
(1) $\colon$ Ở đây đẹp quá!
(2) $\colon3+2=5$.
(3) $\colon8$ là số nguyên tố.
(4) $\colon$ Số $\pi$ có lớn hơn $3$ hay không?
\choiceTF
{(1) là mệnh đề}
{\True (2) là mệnh đề đúng}
{\True (3) là mệnh đề sai}
{\True (4) không phải là mệnh đề}
\loigiai{
\begin{itemchoice}
\item (Sai) (1) là mệnh đề. (Vì): Vì "Ở đây đẹp quá!" là câu cảm thán, không phải khẳng định, nên (1) không là mệnh đề
\item (Đúng) (2) là mệnh đề đúng. (Vì): Vì " $3+2=5$ " là một khẳng định đúng nên (2) là mệnh đề đúng
\item (Đúng) (3) là mệnh đề sai. (Vì): Vì $8$ không là số nguyên tố nên " $8$ là số nguyên tố" là một khẳng định sai, do đó (3) là mệnh đề sai
\item (Đúng) (4) không phải là mệnh đề. (Vì): Vì "Số $\pi$ có lớn hơn $3$ hay không?" là câu hỏi, không phải khẳng định, nên (4) không là mệnh đề
\end{itemchoice}
}
\end{ex}

\begin{ex}
% ID: T10C1B1-01-TN
% Mức: NB
Mệnh đề phủ định của mệnh đề « $15$ là bội số của $3$» là
\choice
{\True $15$ không là bội số của $3$}
{$3$ là bội số của $15$}
{$15$ là ước số của $3$}
{$15$ là số nguyên tố}
\loigiai{
Mệnh đề đã cho là $P$: « $15$ là bội số của $3$».
Mệnh đề này có dạng « $a$ là bội số của $b$».
Mệnh đề phủ định của mệnh đề « $a$ là bội số của $b$» là « $a$ không là bội số của $b$».
Áp dụng vào mệnh đề $P$, mệnh đề phủ định của $P$ là « $15$ không là bội số của $3$».
So sánh với các phương án, ta thấy phương án $A$ trùng với mệnh đề phủ định tìm được.
}
\end{ex}

Hãy xác định tính đúng sai của mỗi mệnh đề sau:
\choiceTF
{« Năm $2025$ là năm nhuận dương lịch»}
{\True « $31$ là số nguyên tố»}
{« Mùa xuân bắt đầu từ tháng $6$ và kết thúc vào tháng $9$»}
{\True « Hình thoi là tứ giác có bốn cạnh bằng nhau»}

\begin{itemchoice}
\item (Sai) « Năm $2025$ là năm nhuận dương lịch». (Vì): Năm $2025$ là năm nhuận dương lịch là là mệnh đề sai vì năm $2025$ không chia hết cho $4$
\item (Đúng) « $31$ là số nguyên tố». (Vì): $31$ là số nguyên tố là mệnh đề đúng vì $31$ chỉ chia hết cho $1$ và chính nó
\item (Sai) « Mùa xuân bắt đầu từ tháng $6$ và kết thúc vào tháng $9$». (Vì): Mùa xuân bắt đầu từ tháng $6$ và kết thúc vào tháng $9$ là mệnh đề sai vì mùa xuân bắt đầu từ tháng $1$ và kết thúc vào tháng $3$
\item (Đúng) « Hình thoi là tứ giác có bốn cạnh bằng nhau». (Vì): Hình thoi là hình có bốn cạnh bằng nhau là mệnh đề đúng
\end{itemchoice}
}
\end{ex}

\dangbt{Nhận biết mệnh đề và mệnh đề chứa biến}
\begin{ex}
% ID: T10C1B1-02-TLN
% Mức: VD
Cho $P\left( n \right)= n^2-6n+10$ với $n$ là số tự nhiên. Có bao nhiêu giá trị của $n$ để $\dfrac{2P\left( n \right)-1}{n-3}$ là số nguyên?
\shortans{$2$}
\loigiai{
$P\left( n \right)= n^2-6n+10\dfrac{2P\left( n \right)-1}{n-3}=\dfrac{2\left( n^2-6n+10 \right)-1}{n-3}=\dfrac{2\left( n^2-6n+9 \right)+2-1}{n-3}=\dfrac{2\left( n-3 \right)^2+1}{n-3}=2\left( n-3 \right)+\dfrac{1}{n-3}$
Để $\dfrac{2P\left( n \right)-1}{n-3}$ là số nguyên thì $n-3$ phải là ước của $1$.
$n-3 \in \{-1; 1\}n-3=1 \Rightarrow n=4n-3=-1 \Rightarrow n=2$
Có $2$ giá trị của $n$.
2
}
\end{ex}

\begin{ex}
% ID: T10C1B1-13-TN
% Mức: TH
Cho mệnh đề chứa biến $P(x)\colon «x+15 \le x^2,;x \in \mathbb{R}$». Trong các mệnh đề sau, mệnh đề nào đúng?
\choice
{Mệnh đề $P(0)$}
{Mệnh đề $P(4)$}
{Mệnh đề $\forall x \in \mathbb{R}, P(x)$}
{\True Mệnh đề $P(-4)$}
\loigiai{
Ta lần lượt kiểm tra các mệnh đề:

\begin{itemize}
\item Với $x=0$:
\(
	P(0)\colon 0+15\le0^2\Leftrightarrow15\le0
	\)
là mệnh đề sai.

\item Với $x=4$:
\[
P(4)\colon 4+15\le4^2\Leftrightarrow19\le16
\]
là mệnh đề sai.

\item Mệnh đề $\forall x\in\mathbb{R},P(x)$ là mệnh đề sai vì $P(0)$ sai.

\item Với $x=-4$:
\[
P(-4)\colon -4+15\le(-4)^2\Leftrightarrow11\le16
\]
là mệnh đề đúng.

\end{itemize}

Vậy mệnh đề đúng là \textbf{D. Mệnh đề $P(-4)$}.
}
\end{ex}

			Mệnh đề này đúng vì với $x=2$ thì $2^2>2\cdot 2-1$.
		\end{enumerate}
	}
\end{bt}

%=== Câu 36 (ID: 0D1H1-5) ===%
\begin{bt}%[0D1H1-5]
	Sử dụng kí hiệu \lq\lq$\forall$\rq\rq \,để viết mỗi mệnh đề sau và xét xem mệnh đề đó là đúng hay sai, giải thích vì sao.
	\begin{enumerate}
		\item $P\colon$\lq\lq Với mọi số thực $x$, $x^2+1>0$\rq\rq.
		\item $Q\colon$\lq\lq Với mọi số tự nhiên $n$, $n^2+n$ chia hết cho $6$\rq\rq.
	\end{enumerate}
	\loigiai{
		\begin{enumerate}
			\item $P\colon$\lq\lq Với mọi số thực $x$, $x^2+1>0$\rq\rq.\\
			Mệnh đề được viết là $P \colon \lq\lq\forall x \in \mathbb{R}$, $x^2+1>0$\rq\rq.\\
			Xét một số thực $x$ tùy ý, ta phải chứng tỏ rằng $x^2+1>0$.\\
			Thật vậy, ta có $x^2+1 \geq 1>0$.\\
			Vậy mệnh đề $P$ là mệnh đề đúng.
			\item $Q\colon$\lq\lq Với mọi số tự nhiên $n$, $n^2+n$ chia hết cho $6$\rq\rq.\\
			Mệnh đề được viết là $Q\colon\lq\lq \forall n \in \mathbb{N}$, $\left(n^2+n\right) \,\vdots\, 6$\rq\rq.\\
			Để chứng minh mệnh đề $Q$ là sai, ta cần chỉ ra một giá trị cụ thể của $n$ để nhận được mệnh đề sai.\\
			Thật vậy, chọn $n=1$, ta thấy $n^2+n=2$ không chia hết cho $6$.\\
			Vậy mệnh đề $Q$ là mệnh đề sai.
		\end{enumerate}
	}
\end{bt}

%=== Câu 40 (ID: 0D1V1-6) ===%
\begin{bt}%[0D1V1-6]

	\begin{enumerate}[$ (1) $]
		\item \lq\lq $A$ kém $B$ là $ 2 $ tuổi và $A$ hơn $C$ là $ 1 $ tuổi\rq\rq.
		\item \lq\lq $B$ không quá $ 25 $ tuổi và $B, C$ cách nhau $ 3 $ tuổi\rq\rq.
		\item \lq\lq $C$ nhỏ tuổi hơn $A$ và $B$ hơn $A$ là $ 3 $ tuổi\rq\rq.
		\item \lq\lq $A$ là $ 22 $ tuổi hoặc $C$ là $ 25 $ tuổi\rq\rq.
	\end{enumerate}
	Hãy tính tổng số tuổi của $A$, $B$ và $C$.
	\loigiai{
		Do theo $ (1) $ thì $A$ kém hơn $B$ là $ 2 $ tuổi, còn theo $ (3) $ thì $B$ hơn $A$ là $ 3 $ tuổi nên $ (1) $ và $ (3) $ không thể nào cùng đúng.\\
		Theo $ (2) $ thì $B$, $C$ cách nhau $ 3 $ tuổi, còn theo $ (3) $ thì $B$ hơn $A$ là 3 tuổi nên $C$ hơn hoặc bằng tuổi $A$. Trái với $C$ nhỏ tuổi hơn $A$.\\
		Vậy $ (3) $ sai, còn $ (1) $, $ (2) $ và $ (4) $ đúng.\\
		\begin{itemize}
			\item Nếu $A$ là $ 22 $ tuổi thì theo $ (1) $, ta có $B$ là $ 24 $ tuổi và $C$ là $ 21 $ tuổi. Thỏa mãn $ (2) $.
			\item Nếu $C$ là $ 25 $ tuổi thì theo $ (1) $ ta có $A$ là $ 26 $ tuổi và $B$ là $ 28 $ tuổi. Không thỏa mãn $ (2) $.
		\end{itemize}
		Vậy $A$ là $22$ tuổi, $B$ là $24$ tuổi, $C$ là $21$ tuổi nên tổng số tuổi của $A$, $B$, $C$ là $ 67 $ tuổi.
	}
\end{bt}
