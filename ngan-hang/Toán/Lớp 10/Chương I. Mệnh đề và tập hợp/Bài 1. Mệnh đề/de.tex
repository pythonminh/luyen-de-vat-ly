%%%=============EX_1=============%%%
\begin{ex}
	Trong các câu sau, câu nào \textbf{không phải} là mệnh đề chứa biến?
	\choice
		{$x^2-2x-3=0$}
		{$3n+2\vdots 5$}
		{\True Số $1$ có một ước duy nhất là chính nó}
		{$x-5y=10$}
	\loigiai{
		Mệnh đề chứa biến là một câu khẳng định mà trong đó có chứa biến số, và tính đúng sai của nó phụ thuộc vào giá trị của biến số đó.
		Xét các phương án:
		Phương án A:  Đây là một mệnh đề chứa biến $x$. Tính đúng sai của nó phụ thuộc vào giá trị của $x$.
		Phương án B:  Đây là một mệnh đề chứa biến $n$. Tính đúng sai của nó phụ thuộc vào giá trị của $n$.
		Phương án C:  Đây là một câu khẳng định cụ thể, không chứa biến. Tính đúng sai của nó không phụ thuộc vào bất kỳ biến nào (câu này là một mệnh đề đúng).
		Phương án D:  Đây là một mệnh đề chứa hai biến $x, y$. Tính đúng sai của nó phụ thuộc vào giá trị của $x$ và $y$.
		Vậy, câu không phải là mệnh đề chứa biến là phương án C.
		}
\end{ex}

		Vậy, khẳng định đúng là B.
		}

\begin{ex}
	Phủ định của mệnh đề ``$\forall x \in \mathbb{R}, x^2 - 2x + 1 \ge 0$'' là mệnh đề nào sau đây?
	\choice
		{\True $\exists x \in \mathbb{R}, x^2 - 2x + 1 < 0$}
		{ $\exists x \in \mathbb{R}, x^2 - 2x + 1 > 0$}
		{ $\forall x \in \mathbb{R}, x^2 - 2x + 1 < 0$}
		{ $\forall x \in \mathbb{R}, x^2 - 2x + 1 \le 0$}
	\loigiai{
		Mệnh đề đã cho là $P\colon \forall x \in \mathbb{R}, x^2 - 2x + 1 \ge 0$.
		Phủ định của mệnh đề $P$ được ký hiệu là $\overline{P}$.
		Quy tắc phủ định của mệnh đề lượng từ:
		- Phủ định của $\forall$ là $\exists$.
		- Phủ định của $\ge$ là $<$.
		Vậy, phủ định của mệnh đề $P$ là $\overline{P}\colon \exists x \in \mathbb{R}, x^2 - 2x + 1 < 0$.
		}
\end{ex}

\begin{ex}
	Cho các mệnh đề sau:
	\choiceTF
		{1. Mệnh đề ``$\exists n \in \mathbb{N}, n^2 - 1$ chia hết cho $3$'' là đúng.}
		{2. Mệnh đề ``$\forall x \in \mathbb{R}, x^2 + 1 > 0$'' là đúng.}
		{3. Mệnh đề ``$\exists x \in \mathbb{Z}, x^2 = 5$'' là đúng.}
		{4. Mệnh đề ``$\forall k \in \mathbb{Z}, k^2 + k + 1$ là số lẻ'' là đúng.}
	\loigiai{
		Xét từng mệnh đề:
		Câu 1. Mệnh đề ``$\exists n \in \mathbb{N}, n^2 - 1$ chia hết cho $3$''.
		Nếu $n=2$, $n^2-1 = 2^2-1 = 3$, chia hết cho $3$.
		Vậy mệnh đề này là đúng.
		Câu 2. Mệnh đề ``$\forall x \in \mathbb{R}, x^2 + 1 > 0$''.
		Với mọi số thực $x$, $x^2 \ge 0$. Do đó, $x^2+1 \ge 1$.
		Vì $1 > 0$, nên $x^2+1 > 0$ với mọi $x \in \mathbb{R}$.
		Vậy mệnh đề này là đúng.
		Câu 3. Mệnh đề ``$\exists x \in \mathbb{Z}, x^2 = 5$''.
		Phương trình $x^2=5$ có nghiệm $x = \pm\sqrt{5}$.
		$\sqrt{5}$ không phải là số nguyên.
		Vậy không có số nguyên $x$ nào thỏa mãn $x^2=5$. Mệnh đề này là sai.
		Câu 4. Mệnh đề ``$\forall k \in \mathbb{Z}, k^2 + k + 1$ là số lẻ''.
		Ta có $k^2+k = k(k+1)$. Đây là tích của hai số nguyên liên tiếp, nên $k(k+1)$ luôn là số chẵn.
		Do đó, $k^2+k+1 = (\text{số chẵn}) + 1 = \text{số lẻ}$.
		Vậy mệnh đề này là đúng.
		}
\end{ex}

	Cho mệnh đề $P(n)\colon n^2+n+1$ là số nguyên tố.
	\choiceTF
		{1. Mệnh đề ``$\exists n \in \mathbb{N}, P(n)$'' là đúng.}
		{2. Mệnh đề ``$\forall n \in \mathbb{N}, P(n)$'' là đúng.}
	\loigiai{
		Câu 1. Mệnh đề ``$\exists n \in \mathbb{N}, P(n)$''.
		Ta cần tìm một số tự nhiên $n$ sao cho $n^2+n+1$ là số nguyên tố.
		Nếu $n=1$, $1^2+1+1 = 3$, là số nguyên tố.
		Vậy mệnh đề ``$\exists n \in \mathbb{N}, P(n)$'' là đúng.
		Câu 2. Mệnh đề ``$\forall n \in \mathbb{N}, P(n)$''.
		Ta cần kiểm tra xem với mọi số tự nhiên $n$, $n^2+n+1$ có phải là số nguyên tố hay không.
		Nếu $n=1$, $1^2+1+1 = 3$ (nguyên tố).
		Nếu $n=2$, $2^2+2+1 = 7$ (nguyên tố).
		Nếu $n=3$, $3^2+3+1 = 13$ (nguyên tố).
		Nếu $n=4$, $4^2+4+1 = 21 = 3 \times 7$ (không phải số nguyên tố).
		Vì có một trường hợp $n=4$ mà $P(4)$ sai, nên mệnh đề ``$\forall n \in \mathbb{N}, P(n)$'' là sai.
		}
\end{ex}

\begin{ex}
	Viết phủ định của mệnh đề sau và xét tính đúng sai của mệnh đề phủ định đó:
	$P\colon \exists x \in \mathbb{R}, x^2 + x + 1 = 0$.
	\shortans{Phủ định: $\forall x \in \mathbb{R}, x^2 + x + 1 \ne 0$. Mệnh đề phủ định là đúng.}
\loigiai{
	Mệnh đề đã cho là $P\colon \exists x \in \mathbb{R}, x^2 + x + 1 = 0$.
	Phủ định của mệnh đề $P$ là $\overline{P}\colon \forall x \in \mathbb{R}, x^2 + x + 1 \ne 0$.
	
	Để xét tính đúng sai của $\overline{P}$, ta xét phương trình $x^2 + x + 1 = 0$.
	Tính biệt thức $\Delta = b^2 - 4ac = 1^2 - 4(1)(1) = 1 - 4 = -3$.
	Vì $\Delta < 0$, và hệ số $a=1 > 0$, nên tam thức bậc hai $x^2+x+1$ luôn dương với mọi $x \in \mathbb{R}$.
	Tức là $x^2 + x + 1 > 0$ với mọi $x \in \mathbb{R}$.
	Do đó, $x^2 + x + 1 \ne 0$ với mọi $x \in \mathbb{R}$.
	Vậy mệnh đề phủ định $\overline{P}\colon \forall x \in \mathbb{R}, x^2 + x + 1 \ne 0$ là một mệnh đề đúng.
	}
\end{ex}

	- ``Bình phương của nó'': $x^2$.
	- ``Không âm'': $\ge 0$.

	- Phủ định của ``mọi'' là ``tồn tại ít nhất một''.

	}

	C.  $R$: "$\exists x \in \mathbb{R}, x + \dfrac{1}{x} \ge 2$";
	D.  $S$: "$\exists x \in \mathbb{R}, x^2 - x + 1 < 0$".

	D.  $\overline{S}$: "$\forall x \in \mathbb{R}, x^2 - x + 1 \ge 0$".
	Xét biểu thức $x^2 - x + 1 = \left(x - \dfrac{1}{2}\right)^2 + \dfrac{3}{4} \ge \dfrac{3}{4} > 0$ đúng with mọi $x \in \mathbb{R}$. Do đó, mệnh đề phủ định $\overline{S}$ là mệnh đề đúng.
