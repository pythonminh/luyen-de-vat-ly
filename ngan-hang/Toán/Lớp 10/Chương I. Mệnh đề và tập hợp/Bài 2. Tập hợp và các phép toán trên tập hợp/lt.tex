\subsubsection{Các khái niệm cơ bản về tập hợp}
\paragraph{Tập hợp}
\begin{hoatdong}{Khởi động}
	Câu lạc bộ Lịch sử có $12$ thành viên (không có hai bạn nào trùng tên), tổ chức hai chuyên đề trên một phần mềm họp trực tuyến. Tên các thành viên tham gia mỗi chuyên đề được hiển thị trên màn hình:
	\begin{itemize}
		\item Chuyên đề 1: Nam, Tú, Khánh, Hương, Bình, Chi, Ngân.
		\item Chuyên đề 2: Hương, Khánh, Hiển, Chi, Bình, Lam, Tú, Hân.
		\item Có bao nhiêu thành viên vắng mặt trong cả hai chuyên đề nhỉ?
	\end{itemize}
\end{hoatdong}
\begin{hoatdong}{Hoạt động 1}
	Trong tình huống mở đầu, gọi $A$ là tập hợp những thành viên tham gia Chuyên đề 1, $B$ là tập hợp những thành viên tham gia Chuyên đề 2.
	\begin{itemize}
		\item Nam có là một phần tử của tập hợp $A$ không? Ngân có là một phần tử của tập hợp $B$ không?
		\item Hãy mô tả các tập hợp $A$ và $B$ bằng cách liệt kê các phần tử.
	\end{itemize}
\end{hoatdong}
\begin{hoatdong}{Hoạt động 2}
	Cho tập hợp $C = \{\text{châu Á}; \text{châu Âu}; \text{châu Đại Dương}; \text{châu Mỹ}; \text{châu Nam Cực}; \text{châu Phi}\}$.
	\begin{itemize}
		\item Hãy chỉ ra tính chất đặc trưng cho các phần tử của tập hợp $C$.
		\item Tập hợp $C$ có bao nhiêu phần tử?
	\end{itemize}
\end{hoatdong}
\begin{itemize}
	\item Để chỉ ra một phần tử $a$ thuộc tập hợp $S$, ta viết $a \in S$ (đọc là $a$ thuộc $S$).
	\item Để chỉ ra một phần tử $a$ không thuộc tập hợp $S$, ta viết $a \notin S$ (đọc là $a$ không thuộc $S$).
	\item Có thể mô tả một tập hợp bằng một trong hai cách sau:
	\begin{itemize}
		\item Cách 1: Liệt kê các phần tử của tập hợp;
		\item Cách 2: Chỉ ra tính chất đặc trưng cho các phần tử của tập hợp.
	\end{itemize}
	\item Số phần tử của tập hợp $S$ được kí hiệu là $n(S)$.
	\item Tập hợp không chứa phần tử nào được gọi là tập rỗng, kí hiệu là $\emptyset$.
\end{itemize}
\begin{vd}
	Cho $D = \{n \in \mathbb{N} \mid n \text{ là số nguyên tố}, 5 < n < 20\}$.
	\begin{itemize}
		\item[a)] Dùng kí hiệu $\in, \notin$ để viết câu trả lời cho câu hỏi sau: Trong các số $5; 12; 17; 18$, số nào thuộc tập hợp $D$, số nào không thuộc tập hợp $D$?
		\item[b)] Viết tập hợp $D$ bằng cách liệt kê các phần tử. Tập hợp $D$ có bao nhiêu phần tử?
	\end{itemize}
\end{vd}
Vì $5$ không lớn hơn $5$ nên $5 \notin D$. Tương tự, $12 \notin D$, $17 \in D$, $18 \notin D$. Các số nguyên tố lớn hơn $5$ và nhỏ hơn $20$ là $7; 11; 13; 17; 19$. Do đó $D = \{7; 11; 13; 17; 19\}$. Tập hợp $D$ có $5$ phần tử, kí hiệu $n(D) = 5$.
\begin{hoatdong}{Luyện tập 1}
	Gọi $X$ là tập nghiệm của phương trình $x^2 - 24x + 143 = 0$. Các mệnh đề sau đúng hay sai?
	\begin{itemize}
		\item[a)] $13 \in X$;
		\item[b)] $11 \notin X$;
		\item[c)] $n(X) = 2$.
	\end{itemize}
\end{hoatdong}
\paragraph{Tập hợp con}
\begin{hoatdong}{Hoạt động 3}
	Gọi $H$ là tập hợp các bạn tham gia Chuyên đề 2 trong tình huống mở đầu có tên bắt đầu bằng chữ $H$. Các phần tử của tập hợp $H$ có là phần tử của tập hợp $B$ không?
\end{hoatdong}
\begin{itemize}
	\item Nếu mọi phần tử của tập hợp $T$ đều là phần tử của tập hợp $S$ thì ta nói $T$ là một tập hợp con (tập con) của $S$ và viết là $T \subset S$ (đọc là $T$ chứa trong $S$ hoặc $T$ là tập con của $S$).
	\item Thay cho $T \subset S$, ta còn viết $S \supset T$ (đọc là $S$ chứa $T$).
	\item Kí hiệu $T \not\subset S$ để chỉ $T$ không là tập con của $S$.
	\item Từ định nghĩa trên, $T$ là tập con của $S$ nếu mệnh đề sau đúng: $\forall x, x \in T \Rightarrow x \in S$.
	\item Quy ước tập rỗng $\emptyset$ là tập con của mọi tập hợp.
\end{itemize}

		\begin{tikzpicture}
			\draw[line width=0.8pt] (0,0) ellipse (1.8cm and 1.2cm);
			\draw (0,0.8) node {$S$};
			\draw[line width=0.8pt] (0.2,-0.2) ellipse (0.8cm and 0.6cm);
			\draw (0.2,-0.2) node {$T$};
		\end{tikzpicture}%
}}
\begin{vd}
	Cho tập hợp $S = \{2; 3; 5\}$. Những tập hợp nào sau đây là tập con của $S$?
	\begin{center}
		$S_1 = \{3\}$; $S_2 = \{0; 2\}$; $S_3 = \{3; 5\}$.
	\end{center}
\end{vd}
Tập hợp $S_1$ là tập con của $S$ vì phần tử duy nhất của nó là $3$ cũng là phần tử của $S$. Tập hợp $S_2$ không phải là tập con của $S$ vì phần tử $0 \in S_2$ nhưng $0 \notin S$. Tập hợp $S_3$ là tập con của $S$ vì các phần tử $3$ và $5$ đều là phần tử của $S$.
\paragraph{Hai tập hợp bằng nhau}
\begin{hoatdong}{Hoạt động 4}
	Sơn và Thu viết tập hợp các số chính phương nhỏ hơn $100$ như sau:
	\begin{itemize}
		\item Sơn: $S = \{0; 1; 4; 9; 16; 25; 36; 49; 64; 81\}$;
		\item Thu: $T = \{n \in \mathbb{N} \mid n \text{ là số chính phương}, n < 100\}$.
	\end{itemize}
	Hỏi bạn nào viết đúng?
\end{hoatdong}
\begin{itemize}
	\item Hai tập hợp $S$ và $T$ được gọi là hai tập hợp bằng nhau nếu mỗi phần tử của $T$ cũng là phần tử của tập hợp $S$ và ngược lại, kí hiệu là $S = T$.
	\item Nếu $S \subset T$ và $T \subset S$ thì $S = T$.
\end{itemize}
\begin{vd}
	Cho hai tập hợp:
	\begin{itemize}
		\item $C = \{n \in \mathbb{N} \mid n \text{ là bội chung của 2 và 3}, n < 30\}$;
		\item $D = \{n \in \mathbb{N} \mid n \text{ là bội của 6}, n < 30\}$.
	\end{itemize}
	Chứng minh rằng $C = D$.
\end{vd}
Các bội chung của $2$ và $3$ là bội của $6$. Do đó các phần tử của $C$ là các bội của $6$ nhỏ hơn $30$. Viết $C$ dưới dạng liệt kê ta được $C = \{0; 6; 12; 18; 24\}$. Tương tự, tập hợp $D$ gồm các bội của $6$ nhỏ hơn $30$ nên $D = \{0; 6; 12; 18; 24\}$. Vậy $C = D$.
\begin{hoatdong}{Luyện tập 2}
	Giả sử $C$ là tập hợp các hình bình hành có hai đường chéo vuông góc; $D$ là tập hợp các hình vuông. Các mệnh đề sau đúng hay sai?
	\begin{itemize}
		\item[a)] $C \subset D$;
		\item[b)] $C \supset D$;
		\item[c)] $C = D$.
	\end{itemize}
\end{hoatdong}
\subsubsection{Các tập hợp số}
\paragraph{Mối quan hệ giữa các tập hợp số}
\begin{hoatdong}{Hoạt động 5}
	Các mệnh đề sau đúng hay sai?
	\begin{itemize}
		\item[a)] Mọi số nguyên đều viết được dưới dạng phân số;
		\item[b)] Tập hợp các số thực chứa tập hợp các số hữu tỉ;
		\item[c)] Tồn tại một số thực không là số hữu tỉ.
	\end{itemize}
\end{hoatdong}
\begin{itemize}
	\item Tập hợp các số tự nhiên $\mathbb{N} = \{0; 1; 2; 3; \dots\}$.
	\item Tập hợp các số nguyên $\mathbb{Z} = \{\dots; -3; -2; -1; 0; 1; 2; 3; \dots\}$.
	\item Tập hợp các số hữu tỉ $\mathbb{Q}$ gồm các số viết được dưới dạng phân số $\dfrac{a}{b}$, với $a, b \in \mathbb{Z}, b \neq 0$.
	\item Tập hợp các số thực $\mathbb{R}$ gồm các số hữu tỉ và các số vô tỉ.
\end{itemize}
\haicotchay{Mối quan hệ giữa các tập hợp số: $\mathbb{N} \subset \mathbb{Z} \subset \mathbb{Q} \subset \mathbb{R}$.}{\centering

		\begin{tikzpicture}
			\draw[line width=0.8pt] (0,0) ellipse (3cm and 2cm);
			\draw (0,1.3) node {$\mathbb{R}$};
			\draw[line width=0.8pt] (0,-0.2) ellipse (2.2cm and 1.4cm);
			\draw (0,0.7) node {$\mathbb{Q}$};
			\draw[line width=0.8pt] (0,-0.4) ellipse (1.5cm and 0.9cm);
			\draw (0,0.1) node {$\mathbb{Z}$};
			\draw[line width=0.8pt] (0,-0.5) ellipse (0.8cm and 0.5cm);
			\draw (0,-0.5) node {$\mathbb{N}$};
		\end{tikzpicture}%
}}
\begin{vd}
	Hãy xác định tính đúng sai của các mệnh đề sau:
	\begin{itemize}
		\item[a)] $3,274 \in \mathbb{Q}$;
		\item[b)] $\sqrt{2} \in \mathbb{R}$;
		\item[c)] $\dfrac{3}{4} \in \mathbb{Z}$.
	\end{itemize}
\end{vd}
Mệnh đề a) đúng vì $3,274 = \dfrac{3274}{1000}$ là một số hữu tỉ. Mệnh đề b) đúng vì $\sqrt{2}$ là số vô tỉ nên nó cũng là một số thực. Mệnh đề c) sai vì $\dfrac{3}{4}$ không phải là số nguyên.
\begin{hoatdong}{Luyện tập 3}
	Cho tập hợp $C = \{-4; 0; 1; 2\}$. Các mệnh đề sau đúng hay sai?
	\begin{itemize}
		\item[a)] $C$ là tập con của $\mathbb{Z}$;
		\item[b)] $C$ là tập con của $\mathbb{N}$;
		\item[c)] $C$ là tập con của $\mathbb{R}$.
	\end{itemize}
\end{hoatdong}
\paragraph{Các tập con thường dùng của tập số thực}
\begin{hoatdong}{Hoạt động 6}
	Cho hai tập hợp $C = \{x \in \mathbb{R} \mid x \ge 3\}$ và $D = \{x \in \mathbb{R} \mid x > 3\}$. Các mệnh đề sau đúng hay sai?
	\begin{itemize}
		\item[a)] $C, D$ là các tập con của $\mathbb{R}$;
		\item[b)] $\forall x, x \in C \Rightarrow x \in D$;
		\item[c)] $3 \in C$ nhưng $3 \notin D$;
		\item[d)] $C = D$.
	\end{itemize}
\end{hoatdong}
Một số tập con thường dùng của tập số thực $\mathbb{R}$ ($a$ và $b$ là các số thực, $a < b$):
\begin{itemize}
	\item Khoảng:
	\begin{itemize}
		\item $(a; b) = \{x \in \mathbb{R} \mid a < x < b\}$
		\item $(a; +\infty) = \{x \in \mathbb{R} \mid x > a\}$
		\item $(-\infty; b) = \{x \in \mathbb{R} \mid x < b\}$
		\item $(-\infty; +\infty) = \mathbb{R}$
	\end{itemize}
	\item Đoạn:
	\begin{itemize}
		\item $[a; b] = \{x \in \mathbb{R} \mid a \le x \le b\}$
	\end{itemize}
	\item Nửa khoảng:
	\begin{itemize}
		\item $[a; b) = \{x \in \mathbb{R} \mid a \le x < b\}$
		\item $(a; b] = \{x \in \mathbb{R} \mid a < x \le b\}$
		\item $[a; +\infty) = \{x \in \mathbb{R} \mid x \ge a\}$
		\item $(-\infty; b] = \{x \in \mathbb{R} \mid x \le b\}$
	\end{itemize}
	\item Kí hiệu $+\infty$ đọc là dương vô cực (hoặc dương vô cùng), kí hiệu $-\infty$ đọc là âm vô cực (hoặc âm vô cùng).
	\item $a, b$ gọi là các đầu mút của đoạn, khoảng hay nửa khoảng.
\end{itemize}
\begin{vd}
	Viết các tập hợp sau dưới dạng các khoảng, đoạn, nửa khoảng trong $\mathbb{R}$ rồi biểu diễn trên trục số:
	\begin{itemize}
		\item[a)] $C = \{x \in \mathbb{R} \mid 2 \le x \le 7\}$;
		\item[b)] $D = \{x \in \mathbb{R} \mid x < 2\}$.
	\end{itemize}
\end{vd}
Ta có $C = [2; 7]$ và $D = (-\infty; 2)$. Để biểu diễn trên trục số, ta vẽ:
\begin{itemize}
	\item Với $C = [2; 7]$, ta gạch bỏ các phần tử nằm ngoài đoạn $[2; 7]$.
	\item Với $D = (-\infty; 2)$, ta gạch bỏ các phần tử lớn hơn hoặc bằng $2$.
\end{itemize}
\begin{hoatdong}{Luyện tập 4}
	Hãy ghép mỗi dòng ở cột bên trái với một dòng thích hợp ở cột bên phải:
	\begin{itemize}
		\item[1)] $x \in [2; 5]$ ghép với d) $2 \le x \le 5$
		\item[2)] $x \in (2; 5]$ ghép với a) $2 < x \le 5$
		\item[3)] $x \in [7; +\infty)$ ghép với b) $x \ge 7$
		\item[4)] $x \in (7; 10)$ ghép với c) $7 < x < 10$
	\end{itemize}
\end{hoatdong}
\subsubsection{Các phép toán trên tập hợp}
\paragraph{Giao của hai tập hợp}
\begin{hoatdong}{Hoạt động 7}
	Viết tập hợp $X$ gồm những thành viên tham gia cả hai chuyên đề 1 và 2 trong tình huống mở đầu. Tập $X$ có phải là tập con của tập $A$ không? Tập $X$ có phải là tập con của tập $B$ không? ($A, B$ là các tập hợp trong Hoạt động 1).
\end{hoatdong}
\haicotchay{Tập hợp gồm các phần tử thuộc cả hai tập hợp $S$ và $T$ gọi là giao của hai tập hợp $S$ và $T$, kí hiệu là $S \cap T$. $S \cap T = \{x \mid x \in S \text{ và } x \in T\}$.}{\centering
	\resizebox{\linewidth}{!}{%
		\begin{tikzpicture}
			\begin{scope}
				\clip (0,0) circle (1.2cm);
				\fill[gray!30] (1,0) circle (1.2cm);
			\end{scope}
			\draw[line width=0.8pt] (0,0) circle (1.2cm) node[left] {$S$};
			\draw[line width=0.8pt] (1,0) circle (1.2cm) node[right] {$T$};
			\draw (0.5,0) node {$S \cap T$};
		\end{tikzpicture}%
}}
\begin{vd}
	\begin{itemize}
		\item[a)] Cho hai tập hợp $C = \{4; 7; 27\}$ và $D = \{2; 4; 9; 27; 36\}$. Hãy xác định tập hợp $C \cap D$.
		\item[b)] Cho hai tập hợp $E = [1; +\infty)$ và $F = (-\infty; 3]$. Hãy xác định tập hợp $E \cap F$.
	\end{itemize}
\end{vd}
\begin{itemize}
	\item[a)] Giao của hai tập hợp $C$ và $D$ gồm các phần tử vừa thuộc $C$ vừa thuộc $D$. Do đó $C \cap D = \{4; 27\}$.
	\item[b)] Giao của hai tập hợp $E$ và $F$ là tập hợp gồm các số thực $x$ thoả mãn $1 \le x$ và $x \le 3$, tức là $1 \le x \le 3$. Vậy $E \cap F = [1; 3]$.
\end{itemize}
\begin{hoatdong}{Luyện tập 5}
	Cho các tập hợp $C = [1; 5]$, $D = [-2; 3]$. Hãy xác định tập hợp $C \cap D$.
\end{hoatdong}
\paragraph{Hợp của hai tập hợp}
\begin{hoatdong}{Hoạt động 8}
	Trở lại tình huống mở đầu, hãy xác định tập hợp các thành viên tham gia Chuyên đề 1 hoặc Chuyên đề 2.
\end{hoatdong}
\haicotchay{Tập hợp gồm các phần tử thuộc tập hợp $S$ hoặc thuộc tập hợp $T$ gọi là hợp của hai tập hợp $S$ và $T$, kí hiệu là $S \cup T$. $S \cup T = \{x \mid x \in S \text{ hoặc } x \in T\}$.}{\centering
	\resizebox{\linewidth}{!}{%
		\begin{tikzpicture}
			\fill[gray!30] (0,0) circle (1.2cm);
			\fill[gray!30] (1,0) circle (1.2cm);
			\draw[line width=0.8pt] (0,0) circle (1.2cm) node[left] {$S$};
			\draw[line width=0.8pt] (1,0) circle (1.2cm) node[right] {$T$};
			\draw (0.5,0) node {$S \cup T$};
		\end{tikzpicture}%
}}
\begin{vd}
	Cho hai tập hợp $C = \{2; 3; 4; 7\}$ và $D = \{-1; 2; 3; 4; 6\}$. Hãy xác định tập hợp $C \cup D$.
\end{vd}
Hợp của hai tập hợp $C$ và $D$ là tập hợp gồm các phần tử thuộc $C$ hoặc thuộc $D$. Do đó $C \cup D = \{-1; 2; 3; 4; 6; 7\}$.
\paragraph{Hiệu của hai tập hợp, phần bù của một tập con}
\begin{hoatdong}{Hoạt động 9}
	Trở lại tình huống mở đầu, hãy xác định tập hợp các thành viên chỉ tham gia Chuyên đề 1 mà không tham gia Chuyên đề 2.
\end{hoatdong}
\haicotchay{Tập hợp gồm các phần tử thuộc $S$ nhưng không thuộc $T$ gọi là hiệu của $S$ và $T$, kí hiệu là $S \setminus T$. $S \setminus T = \{x \mid x \in S \text{ và } x \notin T\}$.}{\centering
	\resizebox{\linewidth}{!}{%
		\begin{tikzpicture}
			\begin{scope}
				\fill[gray!30] (0,0) circle (1.2cm);
			\end{scope}
			\begin{scope}
				\fill[white] (1,0) circle (1.2cm);
			\end{scope}

\begin{itemize}
	\item Khi $T \subset S$ thì $S \setminus T$ được gọi là phần bù của $T$ trong $S$, kí hiệu là $C_S T$.
\end{itemize}
\begin{vd}
	Cho hai tập hợp $A = (-\infty; 5]$ và $B = (0; +\infty)$. Hãy xác định các tập hợp $A \setminus B$ và $B \setminus A$.
\end{vd}
\begin{itemize}
	\item Tập hợp $A \setminus B$ gồm các số thực $x \in A$ và $x \notin B$, tức là $x \le 5$ và $x \le 0$. Vậy $A \setminus B = (-\infty; 0]$.
	\item Tập hợp $B \setminus A$ gồm các số thực $x \in B$ và $x \notin A$, tức là $x > 0$ và $x > 5$. Vậy $B \setminus A = (5; +\infty)$.
\end{itemize}
\begin{hoatdong}{Luyện tập 6}
	Cho các tập hợp $A = [-3; 3]$ và $B = (0; 5)$. Hãy xác định các tập hợp $A \setminus B$, $B \setminus A$ và $C_{\mathbb{R}} B$.
\end{hoatdong}
\begin{emdahoc}
	\begin{itemize}
		\item Khái niệm tập hợp, phần tử thuộc tập hợp, tập hợp rỗng, tập hợp con, hai tập hợp bằng nhau.
		\item Cách mô tả một tập hợp bằng cách liệt kê phần tử hoặc chỉ ra tính chất đặc trưng.
		\item Các tập hợp số thường gặp $\mathbb{N}, \mathbb{Z}, \mathbb{Q}, \mathbb{R}$ và các tập con của $\mathbb{R}$.
		\item Các phép toán giao, hợp, hiệu của hai tập hợp và phần bù của một tập con.
	\end{itemize}
\end{emdahoc}
\begin{emcothe}
	\begin{itemize}
		\item Nhận biết và biểu diễn các tập hợp số, khoảng, đoạn, nửa khoảng trên trục số.
		\item Tìm giao, hợp, hiệu của các tập hợp và phần bù của tập con trong các bài toán thực tiễn.
		\item Sử dụng biểu đồ Ven để giải quyết các bài toán đếm hoặc các tình huống thực tế liên quan đến tập hợp.
	\end{itemize}
\end{emcothe}
\begin{emcobiet}
	\paragraph{Nghịch lí Russel và lý thuyết tập hợp}
	Vào cuối thế kỉ XIX, nhà toán học Đức Georg Cantor (1845 – 1918) đã sáng lập ra Lý thuyết tập hợp ngây thơ (Naive Set Theory), một bước ngoặt vĩ đại làm thay đổi bộ mặt toán học hiện đại. Tuy nhiên, vào năm 1901, nhà triết học và toán học người Anh Bertrand Russell đã chỉ ra một nghịch lí nổi tiếng mang tên ông: "Xét tập hợp $R$ gồm tất cả các tập hợp không tự chứa chính nó làm phần tử. Hỏi $R$ có chứa chính nó không?". Nếu $R$ không chứa $R$, thì theo định nghĩa, $R$ phải thuộc $R$. Ngược lại, nếu $R$ chứa $R$, thì $R$ lại không thuộc $R$. Sự xuất hiện của nghịch lí này đã dẫn đến một cuộc đại khủng hoảng trong cơ sở toán học và thúc đẩy việc xây dựng Lý thuyết tập hợp tiên đề hệ thống hoá chặt chẽ hơn.
\end{emcobiet}
