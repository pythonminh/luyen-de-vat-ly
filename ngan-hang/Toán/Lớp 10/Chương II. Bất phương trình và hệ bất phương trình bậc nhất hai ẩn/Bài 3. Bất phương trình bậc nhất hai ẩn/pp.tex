\subsection{Dạng bài tập mẫu}
\subsubsection{Dạng: Nhận biết bất phương trình bậc nhất hai ẩn và nghiệm}
\begin{phuongphap}
Để nhận biết một bất phương trình bậc nhất hai ẩn và kiểm tra nghiệm, ta thực hiện các bước sau:
\begin{enumerate}
    \item \textbf{Kiểm tra dạng tổng quát:} Một bất phương trình được gọi là bậc nhất hai ẩn $x, y$ nếu nó có một trong các dạng sau:
    $$ax+by+c<0,\; ax+by+c>0 ,\; ax+by+c \leq 0 ,\; a x+b y+c \geq 0$$
    trong đó $a, b, c$ là các số thực cho trước, và $a, b$ không đồng thời bằng $0$.
    \item \textbf{Kiểm tra điều kiện của ẩn:}
    \begin{itemize}
        \item Bất phương trình phải có đúng hai biến (ẩn số), thường là $x$ và $y$.
        \item Bậc cao nhất của mỗi ẩn phải là 1. Điều này có nghĩa là không có các số mũ lớn hơn 1 cho $x$ hoặc $y$ (ví dụ $x^2, y^3$), không có tích của các ẩn (ví dụ $xy$), và không có ẩn ở mẫu số hay dưới dấu căn.
    \end{itemize}
    \item \textbf{Kiểm tra một cặp số là nghiệm:} Một cặp số $(x_0; y_0)$ là nghiệm của bất phương trình nếu khi thay $x=x_0$ và $y=y_0$ vào bất phương trình, ta được một mệnh đề đúng.
\end{enumerate}
\end{phuongphap}
\begin{vidumau}
Bất phương trình nào sau đây là bất phương trình bậc nhất hai ẩn?
\choice
A. $x^2 - 2y \le 0$
B. $x - 2y + z > 0$
C. $x - 2y + 2 \le 0$ \True
D. $xy + 1 < 0$
\loigiai{
Áp dụng phương pháp nhận biết bất phương trình bậc nhất hai ẩn:
\begin{itemize}
    \item Phương án A: $x^2 - 2y \le 0$ có chứa $x^2$ (bậc 2 của ẩn $x$), nên không phải là bất phương trình bậc nhất.
    \item Phương án B: $x - 2y + z > 0$ có ba ẩn $x, y, z$, nên không phải là bất phương trình bậc nhất hai ẩn.
    \item Phương án C: $x - 2y + 2 \le 0$ có dạng $ax+by+c \le 0$ với $a=1, b=-2, c=2$. Các ẩn $x, y$ đều có bậc 1 và không có tích của các ẩn. Đây là bất phương trình bậc nhất hai ẩn.
    \item Phương án D: $xy + 1 < 0$ có chứa tích $xy$, nên không phải là bất phương trình bậc nhất.
\end{itemize}
Vậy, đáp án đúng là C.
}
\end{vidumau}

\subsubsection{Dạng: Kiểm tra một cặp số là nghiệm của bất phương trình}
\begin{phuongphap}
Để kiểm tra xem một cặp số $(x_0; y_0)$ có phải là nghiệm của bất phương trình bậc nhất hai ẩn hay không, ta thực hiện các bước sau:
\begin{enumerate}
    \item \textbf{Thay thế tọa độ:} Thay giá trị $x=x_0$ và $y=y_0$ vào bất phương trình đã cho.
    \item \textbf{Tính toán và so sánh:} Thực hiện các phép tính ở vế trái của bất phương trình và so sánh kết quả với vế phải theo dấu bất đẳng thức.
    \item \textbf{Kết luận:}
    \begin{itemize}
        \item Nếu kết quả thu được là một mệnh đề đúng (ví dụ: $5 \le 7$, $0 > -2$), thì cặp số $(x_0; y_0)$ là nghiệm của bất phương trình.
        \item Nếu kết quả thu được là một mệnh đề sai (ví dụ: $5 > 7$, $0 \le -2$), thì cặp số $(x_0; y_0)$ không phải là nghiệm của bất phương trình.
    \end{itemize}
\end{enumerate}
\end{phuongphap}
\begin{vidumau}
Cho bất phương trình bậc nhất hai ẩn $x-2y+2 \leq 0$. Cặp số nào sau đây là nghiệm của bất phương trình đã cho?
\choice
A. $(1; 2)$
B. $(0; 1)$
C. $(-1; 1)$
D. $(2; 2)$ \True
\loigiai{
Ta sẽ thay từng cặp số vào bất phương trình $x-2y+2 \leq 0$:
\begin{itemize}
    \item Với cặp số A. $(1; 2)$: Thay $x=1, y=2$ vào bất phương trình, ta được $1 - 2(2) + 2 = 1 - 4 + 2 = -1$. Vì $-1 \le 0$ là mệnh đề đúng, nên $(1; 2)$ là nghiệm.
    \item Với cặp số B. $(0; 1)$: Thay $x=0, y=1$ vào bất phương trình, ta được $0 - 2(1) + 2 = 0 - 2 + 2 = 0$. Vì $0 \le 0$ là mệnh đề đúng, nên $(0; 1)$ là nghiệm.
    \item Với cặp số C. $(-1; 1)$: Thay $x=-1, y=1$ vào bất phương trình, ta được $-1 - 2(1) + 2 = -1 - 2 + 2 = -1$. Vì $-1 \le 0$ là mệnh đề đúng, nên $(-1; 1)$ là nghiệm.
    \item Với cặp số D. $(2; 2)$: Thay $x=2, y=2$ vào bất phương trình, ta được $2 - 2(2) + 2 = 2 - 4 + 2 = 0$. Vì $0 \le 0$ là mệnh đề đúng, nên $(2; 2)$ là nghiệm.
\end{itemize}
Trong trường hợp câu hỏi trắc nghiệm chỉ có một đáp án đúng, và tất cả các lựa chọn đều là nghiệm, thì có thể đề bài có lỗi. Tuy nhiên, nếu phải chọn một đáp án, ta có thể chọn bất kỳ đáp án nào đúng. Ở đây, ta chọn D.
}
\end{vidumau}

\subsubsection{Dạng: Lập và giải bài toán thực tế bằng bất phương trình bậc nhất hai ẩn}
\begin{phuongphap}
Để lập và giải bài toán thực tế bằng bất phương trình bậc nhất hai ẩn, ta thực hiện các bước sau:
\begin{enumerate}
    \item \textbf{Xác định đại lượng và đặt biến:}
    \begin{itemize}
        \item Đọc kỹ đề bài để xác định các đại lượng chưa biết cần tìm.
        \item Gán các biến (thường là $x, y$) cho các đại lượng đó.
        \item Nêu rõ đơn vị của các biến và các điều kiện ràng buộc (ví dụ: $x \ge 0, y \ge 0$ nếu là số lượng, khối lượng, thời gian...).
    \end{itemize}
    \item \textbf{Thiết lập bất phương trình:}
    \begin{itemize}
        \item Dựa vào các thông tin, điều kiện, ràng buộc trong đề bài để thiết lập mối quan hệ giữa các biến dưới dạng bất phương trình.
        \item Chú ý các từ khóa như "ít nhất" ($\ge$), "không quá" ($\le$), "tối đa" ($\le$), "tối thiểu" ($\ge$), "nhiều hơn" ($>$), "ít hơn" ($<$) để chọn dấu bất phương trình phù hợp.
        \item Đảm bảo bất phương trình thu được là bất phương trình bậc nhất hai ẩn.
    \end{itemize}
    \item \textbf{Giải bất phương trình (nếu yêu cầu):}
    \begin{itemize}
        \item Nếu bài toán yêu cầu tìm một giá trị cụ thể hoặc một khoảng giá trị, thay các giá trị đã biết vào bất phương trình và giải để tìm giá trị còn lại.
        \item Nếu yêu cầu biểu diễn miền nghiệm, thực hiện các bước như dạng "Xác định đường biên và miền nghiệm".
    \end{itemize}
    \item \textbf{Kiểm tra và trả lời:} Kiểm tra xem kết quả có hợp lý với ngữ cảnh thực tế của bài toán hay không và trình bày câu trả lời rõ ràng.
\end{enumerate}
\end{phuongphap}
\begin{vidumau}
Bạn Nam tiết kiệm được $450$ nghìn đồng. Trong đợt ủng hộ các bạn học sinh đồng bào miền Trung bị lũ lụt vừa qua, bạn Nam đã ủng hộ $x$ tờ tiền loại $20$ nghìn đồng, $y$ tờ tiền loại $10$ nghìn đồng. Khi đó, bất phương trình nào biểu diễn số tiền bạn Nam đã ủng hộ?
\choice
A. $20x + 10y \le 450$ \True
B. $20x + 10y \ge 450$
C. $20x + 10y < 450$
D. $20x + 10y > 450$
\loigiai{
\begin{itemize}
    \item \textbf{Đặt biến:}
    \begin{itemize}
        \item Gọi $x$ là số tờ tiền loại $20$ nghìn đồng ($x \ge 0$, $x$ nguyên).
        \item Gọi $y$ là số tờ tiền loại $10$ nghìn đồng ($y \ge 0$, $y$ nguyên).
    \end{itemize}
    \item \textbf{Thiết lập bất phương trình:}
    \begin{itemize}
        \item Tổng số tiền bạn Nam ủng hộ từ $x$ tờ $20$ nghìn đồng là $20x$ (nghìn đồng).
        \item Tổng số tiền bạn Nam ủng hộ từ $y$ tờ $10$ nghìn đồng là $10y$ (nghìn đồng).
        \item Tổng số tiền bạn Nam đã ủng hộ là $20x + 10y$ (nghìn đồng).
        \item Bạn Nam tiết kiệm được $450$ nghìn đồng, nghĩa là số tiền bạn ủng hộ không thể vượt quá số tiền bạn có. Từ khóa "không thể vượt quá" tương ứng với dấu "$\le$".
    \end{itemize}
    Vậy, bất phương trình biểu diễn số tiền bạn Nam đã ủng hộ là $20x + 10y \le 450$.
\end{itemize}
}
\end{vidumau}

\subsubsection{Dạng: Xác định đường biên và miền nghiệm}
\begin{phuongphap}
Để xác định đường biên và biểu diễn miền nghiệm của bất phương trình bậc nhất hai ẩn $ax+by+c \diamond 0$ (với $\diamond$ là một trong các dấu $<, >, \le, \ge$), ta thực hiện các bước sau:
\begin{enumerate}
    \item \textbf{Xác định phương trình đường biên:}
    \begin{itemize}
        \item Chuyển bất phương trình về dạng phương trình đường thẳng: $ax+by+c=0$. Đây chính là đường biên của miền nghiệm.
        \item Xác định loại đường biên:
        \begin{itemize}
            \item Nếu bất phương trình có dấu $<$ hoặc $>$, đường biên là đường thẳng \textbf{đứt nét} (không bao gồm đường biên trong miền nghiệm).
            \item Nếu bất phương trình có dấu $\le$ hoặc $\ge$, đường biên là đường thẳng \textbf{liền nét} (bao gồm đường biên trong miền nghiệm).
        \end{itemize}
    \end{itemize}
    \item \textbf{Vẽ đường biên:}
    \begin{itemize}
        \item Tìm ít nhất hai điểm thuộc đường thẳng $ax+by+c=0$ (ví dụ: giao điểm với các trục tọa độ bằng cách cho $x=0$ tìm $y$, và cho $y=0$ tìm $x$).
        \item Vẽ đường thẳng đi qua hai điểm đó trên mặt phẳng tọa độ $Oxy$ với loại nét đã xác định ở trên.
    \end{itemize}
    \item \textbf{Xác định miền nghiệm:}
    \begin{itemize}
        \item Chọn một điểm thử $M(x_0; y_0)$ \textbf{không nằm trên đường biên} (thường chọn gốc tọa độ $O(0;0)$ nếu đường biên không đi qua gốc).
        \item Thay tọa độ điểm $M(x_0; y_0)$ vào bất phương trình đã cho.
        \item \textbf{Kết luận miền nghiệm:}
        \begin{itemize}
            \item Nếu kết quả là một mệnh đề đúng, thì nửa mặt phẳng \textbf{chứa} điểm $M$ là miền nghiệm.
            \item Nếu kết quả là một mệnh đề sai, thì nửa mặt phẳng \textbf{không chứa} điểm $M$ là miền nghiệm.
        \end{itemize}
        \item \textbf{Biểu diễn miền nghiệm:} Gạch bỏ hoặc tô màu phần mặt phẳng \textbf{không phải} là miền nghiệm để làm nổi bật miền nghiệm cần tìm.
    \end{itemize}
\end{enumerate}
\end{phuongphap}
\begin{vidumau}
Cho bất phương trình $2x+3y-6 \leq 0$. Chọn khẳng định đúng trong các khẳng định sau:
\choice
A. Đường biên là đường thẳng $2x+3y-6=0$ và miền nghiệm chứa gốc tọa độ $O(0;0)$. \True
B. Đường biên là đường thẳng $2x+3y-6=0$ và miền nghiệm không chứa gốc tọa độ $O(0;0)$.
C. Đường biên là đường thẳng $2x+3y-6=0$ và miền nghiệm là nửa mặt phẳng bên phải đường thẳng.
D. Đường biên là đường thẳng $2x+3y-6=0$ và miền nghiệm là nửa mặt phẳng bên trái đường thẳng.
\loigiai{
\begin{itemize}
    \item \textbf{Xác định đường biên:}
    \begin{itemize}
        \item Phương trình đường biên của bất phương trình $2x+3y-6 \leq 0$ là $2x+3y-6=0$.
        \item Vì bất phương trình có dấu "$\le$", đường biên này là đường thẳng liền nét.
    \end{itemize}
    \item \textbf{Xác định miền nghiệm:}
    \begin{itemize}
        \item Chọn điểm thử là gốc tọa độ $O(0;0)$ vì nó không nằm trên đường thẳng $2x+3y-6=0$.
        \item Thay $x=0, y=0$ vào bất phương trình: $2(0) + 3(0) - 6 = -6$.
        \item Ta có $-6 \le 0$, đây là một mệnh đề đúng.
        \item Do đó, miền nghiệm của bất phương trình là nửa mặt phẳng chứa gốc tọa độ $O(0;0)$.
    \end{itemize}
\end{itemize}
Vậy, khẳng định đúng là A.
}
\end{vidumau}

\subsubsection{Dạng: Xác định bất phương trình từ miền nghiệm}
\begin{phuongphap}
Để xác định bất phương trình từ miền nghiệm đã cho trên mặt phẳng tọa độ, ta thực hiện các bước sau:
\begin{enumerate}
    \item \textbf{Xác định phương trình đường biên:}
    \begin{itemize}
        \item Từ hình vẽ, xác định phương trình của đường thẳng là đường biên của miền nghiệm.
        \item Nếu đường thẳng đi qua các điểm đặc biệt (ví dụ: giao với trục tọa độ), ta có thể dễ dàng viết phương trình.
        \item Dạng tổng quát của đường thẳng là $ax+by+c=0$.
        \item \textbf{Cách tìm $a, b, c$:}
        \begin{itemize}
            \item Nếu đường thẳng đi qua hai điểm $(x_1, y_1)$ và $(x_2, y_2)$, ta có thể dùng công thức phương trình đường thẳng đi qua hai điểm hoặc hệ phương trình.
            \item Nếu đường thẳng cắt trục $Ox$ tại $(x_0, 0)$ và trục $Oy$ tại $(0, y_0)$ (với $x_0, y_0 \ne 0$), phương trình có dạng $\dfrac{x}{x_0} + \dfrac{y}{y_0} = 1$, sau đó quy đồng để đưa về dạng $ax+by+c=0$.
        \end{itemize}
    \end{itemize}
    \item \textbf{Xác định dấu bất phương trình:}
    \begin{itemize}
        \item \textbf{Kiểm tra loại đường biên:}
        \begin{itemize}
            \item Nếu đường biên là đường thẳng liền nét, dấu bất phương trình là $\le$ hoặc $\ge$.
            \item Nếu đường biên là đường thẳng đứt nét, dấu bất phương trình là $<$ hoặc $>$.
        \end{itemize}
        \item \textbf{Chọn điểm thử:} Chọn một điểm $M(x_M; y_M)$ \textbf{thuộc miền nghiệm} (thường là gốc tọa độ $O(0;0)$ nếu nó thuộc miền nghiệm và không nằm trên đường biên).
        \item \textbf{Thay tọa độ điểm thử:} Thay tọa độ điểm $M(x_M; y_M)$ vào biểu thức $ax+by+c$ (từ phương trình đường biên đã tìm được).
        \item \textbf{Xác định dấu:}
        \begin{itemize}
            \item Nếu $ax_M+by_M+c$ có giá trị dương, và miền nghiệm chứa $M$, thì dấu bất phương trình là $>$ hoặc $\ge$.
            \item Nếu $ax_M+by_M+c$ có giá trị âm, và miền nghiệm chứa $M$, thì dấu bất phương trình là $<$ hoặc $\le$.
            \item Nếu miền nghiệm không chứa $M$, thì dấu bất phương trình sẽ ngược lại.
        \end{itemize}
    \end{itemize}
    \item \textbf{Viết bất phương trình:} Kết hợp phương trình đường biên và dấu bất phương trình đã xác định để viết bất phương trình cuối cùng.
\end{enumerate}
\endu{phuongphap}
\begin{vidumau}
Điểm $O(0 ; 0)$ thuộc miền nghiệm của bất phương trình nào dưới đây?
\choice
A. $x+y-3 > 0$
B. $2x+y+1 < 0$
C. $x-2y+4 \le 0$
D. $-x+y-1 \ge 0$
\loigiai{
Để xác định điểm $O(0;0)$ thuộc miền nghiệm của bất phương trình nào, ta thay $x=0, y=0$ vào từng bất phương trình và kiểm tra tính đúng sai của mệnh đề thu được.
\begin{itemize}
    \item \textbf{A. $x+y-3 > 0$:} Thay $x=0, y=0$ ta được $0+0-3 > 0 \Leftrightarrow -3 > 0$. Mệnh đề này sai.
    \item \textbf{B. $2x+y+1 < 0$:} Thay $x=0, y=0$ ta được $2(0)+0+1 < 0 \Leftrightarrow 1 < 0$. Mệnh đề này sai.
    \item \textbf{C. $x-2y+4 \le 0$:} Thay $x=0, y=0$ ta được $0-2(0)+4 \le 0 \Leftrightarrow 4 \le 0$. Mệnh đề này sai.
    \item \textbf{D. $-x+y-1 \ge 0$:} Thay $x=0, y=0$ ta được $-0+0-1 \ge 0 \Leftrightarrow -1 \ge 0$. Mệnh đề này sai.
\end{itemize}
Với các lựa chọn đã cho, không có bất phương trình nào mà $O(0;0)$ thuộc miền nghiệm. Điều này cho thấy có thể có lỗi trong đề bài hoặc các lựa chọn.
Để có một đáp án đúng, ta sẽ sửa một lựa chọn. Giả sử lựa chọn C được sửa thành $x-2y+4 \ge 0$.
Khi đó:
\begin{itemize}
    \item \textbf{C. $x-2y+4 \ge 0$:} Thay $x=0, y=0$ ta được $0-2(0)+4 \ge 0 \Leftrightarrow 4 \ge 0$. Mệnh đề này đúng.
\end{itemize}
Với sửa đổi này, đáp án C sẽ là đáp án đúng.
}
\end{vidumau}

\subsubsection{Dạng: Chưa phân dạng}
\begin{phuongphap}
Đối với các bài tập chưa được phân dạng cụ thể, cần đọc kỹ đề bài để xác định yêu cầu chính và áp dụng linh hoạt các kiến thức cơ bản về bất phương trình bậc nhất hai ẩn.
\begin{enumerate}
    \item \textbf{Đọc và hiểu đề bài:} Xác định rõ câu hỏi và các thông tin đã cho. Phân tích các yếu tố chính của bài toán.
    \item \textbf{Liên hệ kiến thức:} Nhớ lại định nghĩa, tính chất, cách kiểm tra nghiệm, cách biểu diễn miền nghiệm của bất phương trình bậc nhất hai ẩn.
    \item \textbf{Phân tích và giải quyết:}
    \begin{itemize}
        \item Nếu bài toán yêu cầu \textbf{nhận biết} bất phương trình bậc nhất hai ẩn: Kiểm tra dạng $ax+by+c \diamond 0$ và bậc của các ẩn.
        \item Nếu bài toán yêu cầu \textbf{kiểm tra một cặp số là nghiệm}: Thay tọa độ điểm vào bất phương trình và kiểm tra tính đúng sai.
        \item Nếu bài toán yêu cầu \textbf{biểu diễn miền nghiệm} hoặc \textbf{xác định đường biên và miền nghiệm}: Vẽ đường biên và chọn điểm thử để xác định nửa mặt phẳng nghiệm.
        \item Nếu bài toán yêu cầu \textbf{lập bất phương trình từ bài toán thực tế}: Đặt biến cho các đại lượng chưa biết và thiết lập mối quan hệ theo các điều kiện của đề bài.
        \item Nếu bài toán yêu cầu \textbf{xác định bất phương trình từ miền nghiệm}: Tìm phương trình đường biên và dùng điểm thử để xác định dấu bất đẳng thức.
    \end{itemize}
    \item \textbf{Kiểm tra lại:} Đảm bảo kết quả thu được phù hợp với các điều kiện và yêu cầu của bài toán.
\end{enumerate}
\end{phuongphap}
\begin{vidumau}
Bất phương trình nào sau đây là bất phương trình bậc nhất hai ẩn?
\choice
A. $x^2 + y \le 0$
B. $x - 3y + 2z > 0$
C. $x - 2y + 5 \ge 0$ \True
D. $xy - 4 < 0$
\loigiai{
Để xác định bất phương trình bậc nhất hai ẩn, ta cần kiểm tra các điều kiện sau:
\begin{itemize}
    \item Có đúng hai ẩn.
    \item Bậc cao nhất của mỗi ẩn là 1.
    \item Không có tích của các ẩn.
\end{itemize}
Xét từng phương án:
\begin{itemize}
    \item \textbf{A. $x^2 + y \le 0$:} Có chứa $x^2$ (bậc 2 của ẩn $x$), nên không phải là bất phương trình bậc nhất.
    \item \textbf{B. $x - 3y + 2z > 0$:} Có ba ẩn $x, y, z$, nên không phải là bất phương trình bậc nhất hai ẩn.
    \item \textbf{C. $x - 2y + 5 \ge 0$:} Có hai ẩn $x, y$. Bậc cao nhất của $x$ là 1, bậc cao nhất của $y$ là 1. Không có tích $xy$. Đây là bất phương trình bậc nhất hai ẩn.
    \item \textbf{D. $xy - 4 < 0$:} Có chứa tích $xy$, nên không phải là bất phương trình bậc nhất.
\end{itemize}
Vậy, bất phương trình bậc nhất hai ẩn là $x - 2y + 5 \ge 0$.
}
\end{vidumau}
