\dangbt{Nhận biết tích của một vectơ với một số}
\begin{ex}
% ID: AIQ_5B0A2B5569F825
% Mức: VDC
Cho tam giác $ABC$ có trọng tâm $G$. Biết $\vec{GA} = \vec{a}$, $\vec{GB} = \vec{b}$. Hỏi vectơ $\vec{GC}$ bằng vectơ nào sau đây?
\choice
{\True $-\vec{a} - \vec{b}$}
{$\vec{a} + \vec{b}$}
{$\vec{a} - \vec{b}$}
{$-\vec{a} + \vec{b}$}
\loigiai{
Vì $G$ là trọng tâm của tam giác $ABC$ nên $\vec{GA} + \vec{GB} + \vec{GC} = \vec{0}$. Suy ra $\vec{GC} = -(\vec{GA} + \vec{GB}) = -(\vec{a} + \vec{b}) = -\vec{a} - \vec{b}$. Vậy đáp án đúng là A.
}
\end{ex}
