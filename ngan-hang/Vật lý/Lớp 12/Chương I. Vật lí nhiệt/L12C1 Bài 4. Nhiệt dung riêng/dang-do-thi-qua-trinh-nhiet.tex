	\immini[thm]{Khi làm thí nghiệm đun nóng một chất. Kết quả thí
		nghiệm, một học sinh vẽ được biểu diễn sự thay đổi
		nhiệt độ của chất đó theo thời gian như hình bên.
		\choiceTF
		{Tại thời điểm $t_2$, chất đang ở thể lỏng}
		{\True Nhiệt độ nóng chảy của chất là $17^\circ \text{C}$}
		{Tại thời điểm $t_3$, chất đang ở cả thể rắn và thể lỏng}
		{ Nhiệt độ sôi của chất này là $115^\circ \text{C}$}}
	{\begin{tikzpicture}[>=stealth, thick]
			%\draw [dotted] (0,0) grid (6,6);
			\draw [->] (0,0) -- (6.5,0);
			\draw [->] (0,0) -- (0,4.5);
			\draw (0,0) [out = 80, in = -150] to (1,2) -- (3,2) [out = 10, in = -170] to (4.5,4) -- (6,4)
			node [left] at (0,2) {$17$}
			node [left] at (0,4) {$117$}
			node [left] at (0,0) {$O$}
			node [below] at (0.45,0) {$t_1$}
			node [below] at (2,0) {$t_2$}
			node [below] at (3.75,0) {$t_3$}
			node [below] at (5,0) {$t_4$}
			node [right] at (0,4.5) {Nhiệt độ $\left(^\circ \text{C} \right) $}
			node [above] at (6,0) {Thời gian};
			\draw [dashed]
			(0,1.5) -- (0.45,1.5) -- (0.45,0)
			(0,2) -- (2,2) -- (2,0)
			(0,3) -- (3.75,3) -- (3.75,0)
			(0,4) -- (5,4) -- (5,0);
	\end{tikzpicture}}
	\loigiai{
		\begin{itemchoice}
			\itemch Sai. Tại thời điểm $t_2$ chất đang trong quá trình nóng chảy nên tồn tại ở cả thể rắn và thể lỏng. Việc nhận định chất chỉ ở thể lỏng là sai.
			\itemch Đúng. Trong suốt thời gian nóng chảy, nhiệt độ của chất không thay đổi. Dựa vào đồ thị, đoạn nằm ngang đầu tiên tương ứng với quá trình này diễn ra ở $17^\circ \text{C}$, do đó nhiệt độ nóng chảy là $17^\circ \text{C}$.
			\itemch Sai. Tại thời điểm $t_3$, quá trình nóng chảy đã kết thúc và chất đang tiếp tục nhận nhiệt để tăng nhiệt độ. Do đó, lúc này chất chỉ tồn tại ở thể lỏng.
			\itemch Sai. Tương tự như sự nóng chảy, trong quá trình sôi nhiệt độ của chất cũng không đổi. Quan sát đoạn nằm ngang thứ hai trên đồ thị, ta thấy nhiệt độ sôi của chất này phải là $117^\circ \text{C}$, không phải $115^\circ \text{C}$.
		\end{itemchoice}
	}
\end{ex}
