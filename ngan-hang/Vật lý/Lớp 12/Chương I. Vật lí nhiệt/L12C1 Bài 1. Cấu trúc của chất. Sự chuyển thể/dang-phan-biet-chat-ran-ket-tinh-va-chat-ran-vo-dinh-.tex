\begin{enumerate}
    \item \textbf{Về cấu trúc vi mô:}
    \begin{itemize}
        \item \textbf{Chất rắn kết tinh:} Các hạt (phân tử, nguyên tử, ion) liên kết chặt chẽ và sắp xếp theo một trật tự hình học tuần hoàn xác định trong không gian gọi là mạng tinh thể.
        \item \textbf{Chất rắn vô định hình:} Các hạt liên kết chặt chẽ nhưng sắp xếp một cách hỗn độn, ngẫu nhiên và không có trật tự tuần hoàn hình học tạo thành mạng tinh thể.
    \end{itemize}
    \item \textbf{Về nhiệt độ nóng chảy:}
    \begin{itemize}
        \item \textbf{Chất rắn kết tinh:} Có nhiệt độ nóng chảy (hoặc đông đặc) hoàn toàn xác định riêng biệt dưới một áp suất cụ thể. Khi đạt nhiệt độ này, chất bắt đầu chuyển thể đột ngột từ rắn sang lỏng.
        \item \textbf{Chất rắn vô định hình:} Không có nhiệt độ nóng chảy xác định. Khi bị nung nóng, chúng hóa mềm đi dần dần rồi hóa lỏng dần trong một khoảng nhiệt độ rộng.
    \end{itemize}
    \item \textbf{Ví dụ minh họa:}
    \begin{itemize}
        \item Chất rắn kết tinh: Thạch anh, muối ăn, kim cương, hầu hết các kim loại (sắt, đồng, nhôm, chì...), nước đá.
        \item Chất rắn vô định hình: Thủy tinh, nhựa đường, cao su, sôcôla, các chất dẻo.
    \end{itemize}
\end{enumerate}
