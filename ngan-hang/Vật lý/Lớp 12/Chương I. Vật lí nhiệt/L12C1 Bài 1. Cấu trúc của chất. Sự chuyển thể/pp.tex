\subsubsection{Dạng: Mô hình động học phân tử và chuyển động Brown}
\begin{phuongphap}
\begin{enumerate}
    \item \textbf{Xác định yêu cầu của đề bài:} Đề bài thường hỏi về các đặc điểm cơ bản của mô hình động học phân tử (các chất được cấu tạo từ phân tử, phân tử chuyển động không ngừng, giữa các phân tử có lực tương tác) hoặc giải thích hiện tượng chuyển động Brown.
    \item \textbf{Nhớ lại các nguyên lí cơ bản:}
    \begin{itemize}
        \item \textbf{Cấu tạo chất:} Các chất được cấu tạo từ các hạt riêng biệt (phân tử, nguyên tử, ion) có kích thước rất nhỏ.
        \item \textbf{Chuyển động nhiệt:} Các hạt này luôn chuyển động hỗn loạn không ngừng. Nhiệt độ càng cao, chuyển động càng nhanh.
        \item \textbf{Lực tương tác:} Giữa các hạt có lực tương tác (hút và đẩy) phụ thuộc vào khoảng cách.
        \item \textbf{Chuyển động Brown:} Là chuyển động hỗn loạn của các hạt rất nhỏ lơ lửng trong chất lỏng hoặc chất khí, do sự va chạm không cân bằng của các phân tử chất lỏng/khí với hạt đó. Đây là bằng chứng trực tiếp cho sự chuyển động không ngừng của phân tử.
    \end{itemize}
    \item \textbf{Đối chiếu với các phương án:} Chọn phương án mô tả đúng nhất các nguyên lí trên hoặc giải thích đúng hiện tượng được nêu.
\end{enumerate}
\end{phuongphap}
\begin{vidumau}
	Điều nào sau đây là đúng khi nói về mô hình động học phân tử?
	\choice
	{Các phân tử chỉ chuyển động khi có nhiệt độ cao.}
	{Các phân tử luôn đứng yên ở nhiệt độ phòng.}
	{\True Các phân tử luôn chuyển động hỗn loạn không ngừng.}
	{Giữa các phân tử không có lực tương tác.}
	\loigiai{Theo mô hình động học phân tử, các phân tử cấu tạo nên vật chất luôn chuyển động hỗn loạn không ngừng, bất kể nhiệt độ. Nhiệt độ chỉ ảnh hưởng đến tốc độ chuyển động của chúng.}
\end{vidumau}

\subsubsection{Dạng: Đặc điểm và phân loại chất rắn}
\begin{phuongphap}
\begin{enumerate}
    \item \textbf{Xác định loại chất rắn cần phân tích:} Đề bài có thể hỏi về chất rắn kết tinh, chất rắn vô định hình, hoặc so sánh đặc điểm của chúng.
    \item \textbf{Nhớ lại đặc điểm của từng loại:}
    \begin{itemize}
        \item \textbf{Chất rắn kết tinh:}
        \begin{itemize}
            \item Cấu trúc: Các hạt (nguyên tử, ion, phân tử) sắp xếp theo một trật tự hình học nhất định, tạo thành mạng tinh thể tuần hoàn.
            \item Nhiệt độ nóng chảy: Có nhiệt độ nóng chảy xác định (ví dụ: nước đá, kim loại, muối).
            \item Tính chất: Dị hướng (tính chất vật lí khác nhau theo các phương khác nhau) hoặc đẳng hướng (nếu là đơn tinh thể).
        \end{itemize}
        \item \textbf{Chất rắn vô định hình:}
        \begin{itemize}
            \item Cấu trúc: Các hạt sắp xếp một cách lộn xộn, không có trật tự tuần hoàn.
            \item Nhiệt độ nóng chảy: Không có nhiệt độ nóng chảy xác định, mềm dần rồi chảy ra trong một khoảng nhiệt độ (ví dụ: thủy tinh, nhựa đường, cao su).
            \item Tính chất: Đẳng hướng (tính chất vật lí như nhau theo mọi phương).
        \end{itemize}
    \end{itemize}
    \item \textbf{Đối chiếu với các phương án:} Chọn phương án mô tả đúng đặc điểm hoặc sự khác biệt giữa các loại chất rắn.
\end{enumerate}
\end{phuongphap}
\begin{vidumau}
	Với mô hình động học phân tử, sự khác biệt về cấu trúc của chất rắn, chất lỏng, chất khí là do sự khác biệt về
	\choice
	{kích thước của các phân tử}
	{khối lượng của các phân tử}
	{\True lực tương tác và cách sắp xếp của các phân tử}
	{tốc độ chuyển động của các phân tử}
	\loigiai{Sự khác biệt cơ bản giữa các trạng thái rắn, lỏng, khí là do lực tương tác giữa các phân tử và cách sắp xếp của chúng. Trong chất rắn, lực tương tác mạnh và phân tử sắp xếp trật tự; trong chất lỏng, lực tương tác yếu hơn và phân tử sắp xếp không trật tự nhưng vẫn gần nhau; trong chất khí, lực tương tác rất yếu và phân tử chuyển động hỗn loạn, xa nhau.}
\end{vidumau}

\subsubsection{Dạng: Hiện tượng và ứng dụng của sự hóa hơi}
\begin{phuongphap}
\begin{enumerate}
    \item \textbf{Xác định hiện tượng hóa hơi:} Đề bài thường liên quan đến sự bay hơi (xảy ra ở mọi nhiệt độ trên mặt thoáng) hoặc sự sôi (xảy ra ở nhiệt độ sôi trong lòng và trên mặt thoáng).
    \item \textbf{Nhớ lại các yếu tố ảnh hưởng đến tốc độ bay hơi:}
    \begin{itemize}
        \item \textbf{Nhiệt độ:} Nhiệt độ càng cao, tốc độ bay hơi càng nhanh (do động năng phân tử lớn hơn, dễ thoát ra khỏi mặt thoáng).
        \item \textbf{Diện tích mặt thoáng:} Diện tích mặt thoáng càng lớn, tốc độ bay hơi càng nhanh (do có nhiều phân tử ở bề mặt có thể thoát ra).
        \item \textbf{Gió (độ ẩm không khí):} Gió thổi nhanh làm hơi nước bay đi, giảm độ ẩm không khí gần mặt thoáng, tăng tốc độ bay hơi. Độ ẩm không khí càng thấp, tốc độ bay hơi càng nhanh.
        \item \textbf{Bản chất chất lỏng:} Mỗi chất lỏng có tốc độ bay hơi khác nhau ở cùng điều kiện (ví dụ: cồn bay hơi nhanh hơn nước).
    \end{itemize}
    \item \textbf{Giải thích hiện tượng hoặc ứng dụng:} Dựa vào các yếu tố trên để giải thích các hiện tượng thực tế (quần áo khô nhanh hơn khi phơi nắng, gió; mồ hôi bay hơi làm mát cơ thể) hoặc các ứng dụng (làm lạnh bằng cách bay hơi).
\end{enumerate}
\end{phuongphap}
\begin{vidumau}
	Hai chất khí có thể trộn lẫn vào nhau tạo nên một hỗn hợp khí đồng đều là vì
	\choice
	{(1). các phân tử khí chuyển động nhiệt.}
	{(2). hai chất khí đã cho không có phản ứng hoá học với nhau.}
	{(3). giữa các phân tử khí có khoảng trống.}
	{\True (1) và (3) đều đúng.}
	\loigiai{Hiện tượng khuếch tán (trộn lẫn) của các chất khí là do các phân tử khí luôn chuyển động hỗn loạn không ngừng (1) và giữa các phân tử khí có khoảng trống (3) để các phân tử có thể di chuyển xen kẽ vào nhau. Điều kiện không phản ứng hóa học (2) là cần thiết để tạo hỗn hợp đồng đều, nhưng không phải là nguyên nhân vật lí trực tiếp của sự trộn lẫn.}
\end{vidumau}

\subsubsection{Dạng: Quá trình chuyển pha}
\begin{phuongphap}
\begin{enumerate}
    \item \textbf{Xác định quá trình chuyển pha:} Đề bài có thể hỏi về nóng chảy, đông đặc, bay hơi, ngưng tụ, sôi, thăng hoa, hoặc sự phụ thuộc của nhiệt độ vào thời gian trong quá trình chuyển pha.
    \item \textbf{Nhớ lại đặc điểm của từng quá trình:}
    \begin{itemize}
        \item \textbf{Nóng chảy/Đông đặc:} Xảy ra ở nhiệt độ nóng chảy/đông đặc xác định (đối với chất kết tinh). Trong quá trình này, nhiệt độ không đổi dù vẫn nhận/nhả nhiệt.
        \item \textbf{Bay hơi/Ngưng tụ:} Bay hơi xảy ra trên mặt thoáng ở mọi nhiệt độ. Ngưng tụ là quá trình ngược lại.
        \item \textbf{Sôi:} Là quá trình hóa hơi đặc biệt, xảy ra trong lòng và trên mặt thoáng của chất lỏng ở nhiệt độ sôi xác định. Nhiệt độ sôi phụ thuộc vào áp suất bên ngoài.
        \item \textbf{Thăng hoa/Ngưng đọng:} Chuyển trực tiếp từ rắn sang khí hoặc ngược lại.
    \end{itemize}
    \item \textbf{Phân tích đồ thị (nếu có):}
    \begin{itemize}
        \item Đoạn đồ thị dốc lên: Chất đang nhận nhiệt và tăng nhiệt độ (chưa chuyển pha).
        \item Đoạn đồ thị nằm ngang: Chất đang chuyển pha (nóng chảy, sôi, đông đặc, ngưng tụ) ở nhiệt độ không đổi. Nhiệt lượng nhận/nhả trong giai đoạn này là nhiệt ẩn chuyển pha.
    \end{itemize}
    \item \textbf{Giải thích hoặc tính toán:} Dựa vào các đặc điểm trên để giải thích hiện tượng hoặc tính toán nhiệt lượng liên quan.
\end{itemize}
\end{phuongphap}
\begin{vidumau}
	Trong quá trình hóa hơi của một lượng chất lỏng ở nhiệt độ sôi thì
	\choice
	{nhiệt độ của chất lỏng tăng dần.}
	{nhiệt độ của chất lỏng giảm dần.}
	{\True nhiệt độ của chất lỏng không thay đổi.}
	{nhiệt độ của chất lỏng có thể tăng hoặc giảm tùy thuộc vào áp suất.}
	\loigiai{Khi chất lỏng sôi, nó hóa hơi ở một nhiệt độ xác định (nhiệt độ sôi). Trong suốt quá trình sôi, dù chất lỏng vẫn tiếp tục nhận nhiệt, nhiệt độ của nó không thay đổi. Nhiệt lượng nhận được trong giai đoạn này được dùng để phá vỡ các liên kết giữa các phân tử, chuyển chúng từ trạng thái lỏng sang trạng thái hơi, chứ không làm tăng động năng trung bình của các phân tử (tức là không làm tăng nhiệt độ).}
\end{vidumau}

\subsubsection{Dạng: Ôn tập 2}
\begin{phuongphap}
\begin{enumerate}
    \item \textbf{Đọc kỹ câu hỏi:} Xác định yêu cầu chính của câu hỏi, đặc biệt chú ý các từ khóa như "đúng", "sai", "không đúng", "phát biểu nào".
    \item \textbf{Phân tích từng phương án:} Đối với mỗi phương án, liên hệ với các kiến thức đã học về cấu trúc chất, mô hình động học phân tử, các trạng thái của vật chất và quá trình chuyển pha.
    \item \textbf{Kiểm tra tính chính xác:}
    \begin{itemize}
        \item Nếu câu hỏi yêu cầu tìm phát biểu đúng: Chọn phương án hoàn toàn phù hợp với lí thuyết.
        \item Nếu câu hỏi yêu cầu tìm phát biểu sai/không đúng: Chọn phương án mâu thuẫn với lí thuyết hoặc có thông tin không chính xác.
    \end{itemize}
    \item \textbf{Loại trừ các phương án nhiễu:} Các phương án nhiễu thường chứa thông tin gần đúng nhưng có một chi tiết sai lệch, hoặc chỉ đúng trong một trường hợp cụ thể mà không đúng tổng quát.
\end{enumerate}
\end{phuongphap}
\begin{vidumau}
	Câu nào sau đây nói về chuyển động của phân tử là \textbf{không đúng}?
	\choice
	{Các phân tử luôn chuyển động hỗn loạn không ngừng.}
	{Nhiệt độ càng cao, các phân tử chuyển động càng nhanh.}
	{\True Các phân tử chỉ chuyển động khi vật chất ở thể khí.}
	{Chuyển động Brown là bằng chứng trực tiếp về sự chuyển động không ngừng của phân tử.}
	\loigiai{Phát biểu "Các phân tử chỉ chuyển động khi vật chất ở thể khí" là không đúng. Các phân tử luôn chuyển động hỗn loạn không ngừng ở mọi trạng thái (rắn, lỏng, khí), chỉ khác nhau về biên độ và tốc độ chuyển động. Trong chất rắn, phân tử dao động quanh vị trí cân bằng; trong chất lỏng và khí, chúng chuyển động tự do hơn.}
\end{vidumau}

\subsubsection{Dạng: ôn tập bài 1}
\begin{phuongphap}
\begin{enumerate}
    \item \textbf{Tổng hợp kiến thức:} Đề bài dạng ôn tập thường bao quát nhiều nội dung trong bài 1, bao gồm:
    \begin{itemize}
        \item Mô hình động học phân tử (cấu tạo, chuyển động, tương tác).
        \item Các trạng thái của vật chất (rắn, lỏng, khí) và đặc điểm của chúng.
        \item Các quá trình chuyển pha (nóng chảy, đông đặc, bay hơi, sôi, ngưng tụ).
        \item Các yếu tố ảnh hưởng đến sự hóa hơi.
    \end{itemize}
    \item \textbf{Đọc và hiểu rõ câu hỏi:} Xác định chính xác điều mà câu hỏi muốn kiểm tra (ví dụ: tìm phát biểu sai, chọn định nghĩa đúng, giải thích hiện tượng).
    \item \textbf{Đánh giá từng lựa chọn:} Với mỗi lựa chọn, so sánh với các định nghĩa, nguyên lí, và hiện tượng đã học.
    \item \textbf{Chọn đáp án phù hợp:}
    \begin{itemize}
        \item Nếu tìm phát biểu đúng: Chọn lựa chọn hoàn toàn chính xác.
        \item Nếu tìm phát biểu sai: Chọn lựa chọn có thông tin mâu thuẫn với kiến thức vật lí.
    \end{itemize}
    \item \textbf{Cẩn thận với các từ phủ định:} "Không", "chỉ", "luôn luôn", "không bao giờ" có thể thay đổi hoàn toàn ý nghĩa của câu.
\end{enumerate}
\end{phuongphap}
\begin{vidumau}
	Tìm phát biểu \textbf{sai}.
	\choice
	{Các chất được cấu tạo từ các hạt riêng biệt gọi là phân tử.}
	{Các phân tử luôn chuyển động hỗn loạn không ngừng.}
	{Giữa các phân tử có lực tương tác.}
	{\True Lực tương tác giữa các phân tử chỉ là lực hút.}
	\loigiai{Phát biểu "Lực tương tác giữa các phân tử chỉ là lực hút" là sai. Giữa các phân tử có cả lực hút và lực đẩy. Khi khoảng cách giữa các phân tử lớn, lực hút chiếm ưu thế; khi khoảng cách rất nhỏ, lực đẩy chiếm ưu thế. Sự cân bằng giữa hai lực này quyết định trạng thái và tính chất của vật chất.}
\end{vidumau}

\subsubsection{Dạng: Phân loại chất rắn}
\begin{phuongphap}
\begin{enumerate}
\item \textbf{Đọc đề và xác định yêu cầu:} Đề bài thường hỏi về việc phân loại chất rắn thành kết tinh hay vô định hình, hoặc so sánh các tính chất đặc trưng của chúng (ví dụ: có nhiệt độ nóng chảy xác định hay không, cấu trúc hạt).
\item \textbf{Nhớ định nghĩa và đặc điểm cốt lõi:}
    \begin{itemize}
        \item \textbf{Chất rắn kết tinh:}
            \begin{itemize}
                \item Cấu trúc: Các hạt (nguyên tử, ion, phân tử) sắp xếp theo một trật tự hình học nhất định, tạo thành mạng tinh thể tuần hoàn.
                \item Nhiệt độ nóng chảy: Có nhiệt độ nóng chảy (và đông đặc) xác định. Khi nung nóng, chất rắn kết tinh sẽ nóng chảy hoàn toàn tại một nhiệt độ không đổi.
                \item Ví dụ: Kim loại (đồng, sắt), muối ăn, đá quý (kim cương, thạch anh).
            \end{itemize}
        \item \textbf{Chất rắn vô định hình:}
            \begin{itemize}
                \item Cấu trúc: Các hạt sắp xếp một cách lộn xộn, không có trật tự tuần hoàn như mạng tinh thể.
                \item Nhiệt độ nóng chảy: Không có nhiệt độ nóng chảy xác định. Khi nung nóng, chúng mềm dần rồi chảy ra trong một khoảng nhiệt độ.
                \item Ví dụ: Thủy tinh, nhựa đường, cao su, sáp.
            \end{itemize}
    \end{itemize}
\item \textbf{Đối chiếu với các ví dụ và tính chất được nêu:} Không nhầm lẫn giữa "cứng" và "kết tinh". Một số chất vô định hình (như thủy tinh) rất cứng nhưng không có cấu trúc tinh thể.
\end{enumerate}
\end{phuongphap}
\begin{vidumau}
	Chất rắn kết tinh khác chất rắn vô định hình chủ yếu ở chỗ
	\choice
	{\True các hạt sắp xếp tuần hoàn, có nhiệt độ nóng chảy xác định}
	{các hạt chuyển động tự do như trong chất khí}
	{không có lực liên kết giữa các hạt}
	{không có hình dạng và thể tích riêng}
	\loigiai{
Chất rắn kết tinh có đặc điểm nổi bật là các hạt (nguyên tử, ion hoặc phân tử) được sắp xếp một cách trật tự, tuần hoàn trong không gian tạo thành mạng tinh thể. Chính nhờ cấu trúc này mà chất rắn kết tinh có nhiệt độ nóng chảy xác định. Khi đạt đến nhiệt độ nóng chảy, toàn bộ khối chất sẽ chuyển từ thể rắn sang thể lỏng ở nhiệt độ không đổi.
Ngược lại, chất rắn vô định hình không có cấu trúc tinh thể tuần hoàn và do đó không có nhiệt độ nóng chảy xác định, chúng mềm dần khi nung nóng.
}
\end{vidumau}

\subsubsection{Dạng: Nhận biết qua hiện tượng}
\begin{phuongphap}
\begin{enumerate}
\item \textbf{Xác định hiện tượng vật lí được mô tả:} Đề bài sẽ mô tả một hiện tượng (ví dụ: bay hơi, khuếch tán, nén chất khí, giữ hình dạng/thể tích riêng) hoặc một tình huống thực tế.
\item \textbf{Liên hệ hiện tượng với mô hình động học phân tử:}
    \begin{itemize}
        \item \textbf{Bay hơi/Sôi:} Liên quan đến động năng của phân tử, lực liên kết, diện tích mặt thoáng, nhiệt độ. Phân tử có động năng đủ lớn để thắng lực liên kết và thoát ra khỏi bề mặt chất lỏng.
        \item \textbf{Khuếch tán:} Do các phân tử chuyển động hỗn loạn không ngừng và giữa chúng có khoảng trống, cho phép chúng xen kẽ vào nhau.
        \item \textbf{Nén được/Không nén được:} Liên quan đến khoảng cách giữa các phân tử và lực tương tác. Chất khí dễ nén vì khoảng cách phân tử lớn, lực tương tác yếu. Chất lỏng, rắn khó nén vì khoảng cách phân tử nhỏ, lực tương tác mạnh.
        \item \textbf{Giữ hình dạng/thể tích riêng:} Liên quan đến lực liên kết và sự sắp xếp phân tử. Chất rắn có hình dạng và thể tích riêng do lực liên kết mạnh và sắp xếp cố định. Chất lỏng có thể tích riêng nhưng không có hình dạng riêng. Chất khí không có cả hai.
    \end{itemize}
\item \textbf{Giải thích hoặc kết luận theo dữ kiện đề bài:} Dựa vào mối liên hệ trên để giải thích nguyên nhân của hiện tượng hoặc dự đoán kết quả (ví dụ: mặt thoáng lớn $\to$ bay hơi nhanh; nhiệt độ cao $\to$ khuếch tán nhanh).
\end{enumerate}
\end{phuongphap}

\subsubsection{Dạng: Sự bay hơi của chất lỏng và sự biến thiên khối lượng}

\begin{phuongphap}
	\begin{enumerate}
		\item \textbf{Dấu hiệu nhận biết:} Bài toán liên quan đến sự giảm khối lượng của chất lỏng theo thời gian do bay hơi. Có thể yêu cầu tính khối lượng bay hơi, số phân tử bay hơi, hoặc tốc độ bay hơi (số phân tử/giây).
		\item \textbf{Các bước giải cụ thể:}
			\begin{itemize}
				\item \textbf{Tính khối lượng chất lỏng đã bay hơi ($\Delta m$):} Lấy khối lượng ban đầu trừ đi khối lượng còn lại.
				\[ \Delta m = m_{\text{ban đầu}} - m_{\text{còn lại}} \]
				\item \textbf{Tính số mol chất lỏng đã bay hơi ($n$):} Sử dụng khối lượng mol ($M$) của chất.
				\[ n = \dfrac{\Delta m}{M} \]
				\item \textbf{Tính số phân tử đã bay hơi ($\Delta N$):} Sử dụng hằng số Avogadro ($N_A$).
				\[ \Delta N = n \cdot N_A = \dfrac{\Delta m}{M} \cdot N_A \]
				\item \textbf{Tính tốc độ bay hơi ($n_0$ - số phân tử/giây):} Chia tổng số phân tử bay hơi cho tổng thời gian bay hơi ($\Delta \tau$).
				\[ n_0 = \dfrac{\Delta N}{\Delta \tau} \]
			\end{itemize}
		\item \textbf{Lưu ý các bẫy và đổi đơn vị:}
			\begin{itemize}
				\item \textbf{Đơn vị khối lượng:} Đảm bảo khối lượng $\Delta m$ và khối lượng mol $M$ cùng đơn vị (thường là gam).
				\item \textbf{Đơn vị thời gian:} Khi tính tốc độ bay hơi, thời gian $\Delta \tau$ phải được quy đổi về giây ($\mathrm{s}$).
				\item \textbf{Phân biệt:} Khối lượng chất lỏng bị bay hơi $\Delta m$ khác với khối lượng chất lỏng còn lại trong cốc.
			\end{itemize}
	\end{enumerate}
\end{phuongphap}

\begin{vidumau}

			\item Ước tính số phân tử nước thoát ra khỏi bề mặt chất lỏng trung bình trong một giây.
		\end{enumerate}

\subsubsection{Dạng: Bài toán đun nóng nâng nhiệt độ kết hợp nóng chảy và hóa hơi}

\begin{phuongphap}
	\begin{enumerate}
		\item \textbf{Dấu hiệu nhận biết:} Bài toán yêu cầu tính tổng nhiệt lượng cần cung cấp (hoặc tỏa ra) để một khối chất trải qua nhiều giai đoạn: nâng nhiệt độ ở một thể, chuyển thể (nóng chảy/đông đặc, hóa hơi/ngưng tụ), và tiếp tục nâng nhiệt độ ở thể mới.
		\item \textbf{Phân tích các giai đoạn và công thức nhiệt lượng tương ứng:}
			\begin{itemize}
				\item \textbf{Giai đoạn nâng nhiệt độ (không chuyển thể):} Nhiệt lượng $Q$ được tính bằng công thức $Q = m \cdot c \cdot \Delta T$.
					\begin{itemize}
						\item $m$: khối lượng chất (kg).
						\item $c$: nhiệt dung riêng của chất ở thể đó ($\mathrm{J/(kg \cdot K)}$ hoặc $\mathrm{J/(kg \cdot ^\circ C)}$).
						\item $\Delta T$: độ biến thiên nhiệt độ ($\mathrm{K}$ hoặc $^\circ \mathrm{C}$).
					\end{itemize}
				\item \textbf{Giai đoạn chuyển thể (nhiệt độ không đổi):} Nhiệt lượng $Q$ được tính bằng công thức $Q = m \cdot \text{nhiệt ẩn chuyển pha}$.
					\begin{itemize}
						\item \textbf{Nóng chảy/Đông đặc:} $Q = m \cdot \lambda$. ($\lambda$: nhiệt nóng chảy riêng, $\mathrm{J/kg}$).
						\item \textbf{Hóa hơi/Ngưng tụ:} $Q = m \cdot L$. ($L$: nhiệt hóa hơi riêng, $\mathrm{J/kg}$).
					\end{itemize}
			\end{itemize}
		\item \textbf{Tổng hợp nhiệt lượng:} Tổng nhiệt lượng cần thiết là tổng của nhiệt lượng ở tất cả các giai đoạn.
		\[ Q_{\text{tổng}} = \sum Q_i \]
		\item \textbf{Lưu ý các bẫy và đổi đơn vị:}
			\begin{itemize}
				\item \textbf{Nhiệt dung riêng ($c$):} Phải sử dụng đúng nhiệt dung riêng cho từng thể của chất (ví dụ: $c_{\text{nước đá}}$, $c_{\text{nước lỏng}}$, $c_{\text{hơi nước}}$).
				\item \textbf{Nhiệt độ chuyển pha:} Nhiệt độ nóng chảy của nước đá là $0\ ^\circ\mathrm{C}$, nhiệt độ sôi của nước là $100\ ^\circ\mathrm{C}$ (ở áp suất chuẩn).
				\item \textbf{Đơn vị:} Đảm bảo tất cả các đại lượng đều ở đơn vị SI (kg, J, K hoặc $^\circ \mathrm{C}$).
				\item \textbf{Đồ thị:} Nếu có đồ thị, đoạn nằm ngang biểu thị quá trình chuyển pha (nhiệt độ không đổi), đoạn dốc biểu thị quá trình tăng/giảm nhiệt độ.
			\end{itemize}
	\end{enumerate}
\end{phuongphap}

\begin{vidumau}
Một lò nung nung nóng chảy hoàn toàn $2{,}0\ \mathrm{kg}$ nước đá từ nhiệt độ ban đầu $-10\ ^\circ\mathrm{C}$ thành nước lỏng ở $20\ ^\circ\mathrm{C}$. Cho biết nhiệt dung riêng của nước đá là $c_{\text{đ}} = 2100\ \mathrm{J/(kg\cdot K)}$, của nước lỏng là $c_{\text{n}} = 4180\ \mathrm{J/(kg\cdot K)}$ và nhiệt nóng chảy riêng của nước đá là $\lambda = 3{,}34 \cdot 10^5\ \mathrm{J/kg}$.
		\begin{enumerate}
			\item Tính nhiệt lượng $Q_1$ cần thiết để đưa khối nước đá từ $-10\ ^\circ\mathrm{C}$ lên $0\ ^\circ\mathrm{C}$ và nóng chảy hoàn toàn ở $0\ ^\circ\mathrm{C}$.
			\item Tính tổng nhiệt lượng $Q$ cần cung cấp trong toàn bộ quá trình đun để đưa khối nước đá ban đầu thành nước lỏng ở $20\ ^\circ\mathrm{C}$.
		\end{enumerate}
\begin{center}
\resizebox{0.9\linewidth}{!}{%
			\begin{tikzpicture}[scale=1.0, >=stealth]
				\draw[->, thick] (0,-0.8) -- (0,2.5) node[above] {$t\ (^\circ\mathrm{C})$};
				\draw[->, thick] (0,0) -- (4.2,0) node[right] {$\tau$};
				% Draw heating curve
				\draw[thick, blue] (0,-0.5) -- (1,0) -- (2.8,0) -- (3.8,1.2);
				% Marks
				\draw[dashed, gray] (0,-0.5) -- (0,-0.5) node[left, black] {$-10$};
				\node at (0,0) [below left] {$O$};
				\draw[dashed, gray] (3.8,1.2) -- (0,1.2) node[left, black] {$20$};
				\draw[dashed, gray] (3.8,1.2) -- (3.8,0);
				\node at (0.5,-0.6) [font=\tiny, blue] {$\text{Đá}$};
				\node at (1.9,0.2) [font=\tiny, blue] {$\text{Đang chảy}$};
				\node at (3.5,0.7) [font=\tiny, blue] {$\text{Nước}$};
			\end{tikzpicture}%
		}
\end{center}
\loigiai{
\begin{enumerate}
		\item Nhiệt lượng cần thiết để đưa khối nước đá từ $-10\ ^\circ\mathrm{C}$ lên $0\ ^\circ\mathrm{C}$ là:
		\[
		\begin{aligned}
			Q_{\text{nâng}} &= m \cdot c_{\text{đ}} \cdot [0 - (-10)] \\
			&= 2{,}0 \cdot 2100 \cdot 10 \\
			&= 42000\ \mathrm{J}.
		\end{aligned}
		\]
		Nhiệt lượng cần thiết để làm nóng chảy hoàn toàn lượng nước đá này ở $0\ ^\circ\mathrm{C}$ là:
		\[
		\begin{aligned}
			Q_{\text{nóng chảy}} &= \lambda \cdot m \\
			&= 3{,}34 \cdot 10^5 \cdot 2{,}0 \\
			&= 668000\ \mathrm{J}.
		\end{aligned}
		\]
		Nhiệt lượng $Q_1$ cần thiết cho giai đoạn này là:
		\[
		\begin{aligned}
			Q_1 &= Q_{\text{nâng}} + Q_{\text{nóng chảy}} \\
			&= 42000 + 668000 \\
			&= 710000\ \mathrm{J}.
		\end{aligned}
		\]
		\item Sử dụng kết quả nhiệt lượng $Q_1$ ở câu trước, ta tính tiếp nhiệt lượng $Q_2$ để đưa lượng nước lỏng thu được sau khi tan tăng nhiệt độ từ $0\ ^\circ\mathrm{C}$ lên $20\ ^\circ\mathrm{C}$:
		\[
		\begin{aligned}
			Q_2 &= m \cdot c_{\text{n}} \cdot (20 - 0) \\
			&= 2{,}0 \cdot 4180 \cdot 20 \\
			&= 167200\ \mathrm{J}.
		\end{aligned}
		\]
		Tổng nhiệt lượng cần cung cấp trong toàn bộ quá trình là:
		\[
		\begin{aligned}
			Q &= Q_1 + Q_2 \\
			&= 710000 + 167200 \\
			&= 877200\ \mathrm{J}.
		\end{aligned}
		\]
	\end{enumerate}
}
\end{vidumau}
