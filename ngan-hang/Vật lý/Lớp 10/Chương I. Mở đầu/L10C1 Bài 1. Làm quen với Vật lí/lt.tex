\subsection{Lý thuyết}
\subsubsection{Nội dung}
\begin{kienthuc}
\subsubsection{Đối tượng, mục tiêu và phương pháp nghiên cứu của Vật lí}
\paragraph{Đối tượng nghiên cứu}
\begin{itemize}
\item Đối tượng nghiên cứu của Vật lí gồm các dạng vận động của vật chất và năng lượng.
\end{itemize}
\paragraph{Mục tiêu nghiên cứu}
\begin{itemize}
\item Mục tiêu của Vật lí là khám phá ra quy luật tổng quát nhất chi phối sự vận động của vật chất và năng lượng, cũng như tương tác giữa chúng ở mọi cấp độ: vi mô và vĩ mô.
\item Cấp độ vi mô là cấp độ dùng để khảo sát thế giới các hạt cực kì nhỏ bé như hạt nhân, nguyên tử, phân tử.
\item Cấp độ vĩ mô là cấp độ dùng để khảo sát tầm rộng lớn hay rất lớn của vật chất như các vật thể thông thường đến cấp độ vũ trụ.
\item Mục tiêu học tập môn Vật lí giúp học sinh hình thành, phát triển năng lực vật lí.
\end{itemize}
\paragraph{Phương pháp nghiên cứu}
\begin{itemize}
\item Vật lí sử dụng hai phương pháp nghiên cứu chính bổ trợ cho nhau: phương pháp thực nghiệm và phương pháp lí thuyết.
\item Phương pháp thực nghiệm dùng thí nghiệm để phát hiện kết quả mới giúp kiểm chứng, hoàn thiện, bổ sung hay bác bỏ giả thuyết nào đó. Kết quả mới này cần được giải thích bằng lí thuyết đã biết hoặc một lí thuyết mới.
\item Phương pháp lí thuyết sử dụng ngôn ngữ toán học và suy luận lí thuyết để phát hiện một kết quả mới. Kết quả mới này cần được kiểm chứng bằng thực nghiệm.
\item Hai phương pháp hỗ trợ đắc lực cho nhau, trong đó phương pháp thực nghiệm mang tính quyết định.
\end{itemize}
\paragraph{Tiến trình tìm hiểu thế giới tự nhiên dưới góc độ vật lí}
\begin{itemize}
\item Tiến trình nghiên cứu gồm các bước cơ bản sau:
\begin{itemize}
\item Bước 1: Quan sát hiện tượng, suy luận để đề xuất vấn đề cần nghiên cứu.
\item Bước 2: Hình thành giả thuyết nghiên cứu (đưa ra dự đoán khoa học có cơ sở).
\item Bước 3: Thiết kế, xây dựng mô hình kiểm chứng giả thuyết bằng thực nghiệm (thí nghiệm).
\item Bước 4: Thực hiện thí nghiệm, thu thập, phân tích số liệu thực nghiệm.
\item Bước 5: Rút ra kết luận (xác nhận, điều chỉnh hoặc bác bỏ giả thuyết ban đầu).
\end{itemize}
\end{itemize}
\subsubsection{Vật lí với đời sống, khoa học, kĩ thuật và công nghệ}
\paragraph{Ảnh hưởng của Vật lí đến kĩ thuật và công nghệ}
\begin{itemize}
\item Vật lí là cơ sở của khoa học tự nhiên và công nghệ, đóng vai trò nền tảng thúc đẩy các cuộc cách mạng công nghiệp:
\begin{itemize}
\item Cuộc cách mạng công nghiệp lần thứ nhất (thế kỉ XVIII): Thay thế sức lao động thủ công bằng cơ giới hóa nhờ sự phát minh ra động cơ hơi nước dựa trên các nghiên cứu về Nhiệt học.
\item Cuộc cách mạng công nghiệp lần thứ hai (thế kỉ XIX): Đặc trưng bởi sự xuất hiện của các thiết bị dùng điện và năng lượng điện trong mọi lĩnh vực đời sống, sản xuất nhờ các nghiên cứu về Điện từ học.
\item Cuộc cách mạng công nghiệp lần thứ ba (thế kỉ XX): Đặc trưng bởi sự tự động hóa các quá trình sản xuất và công nghệ thông tin nhờ các thành tựu của Vật lí bán dẫn và Vật lí lượng tử.
\item Cuộc cách mạng công nghiệp lần thứ tư (thế kỉ XXI): Phát triển công nghệ siêu nhỏ, trí tuệ nhân tạo, IoT, robot dựa trên các nghiên cứu vật lí hiện đại.
\end{itemize}
\end{itemize}
\paragraph{Ảnh hưởng hai mặt của việc ứng dụng công nghệ}
\begin{itemize}
\item Tích cực: Đem lại cho con người những lợi ích tuyệt vời trong đời sống, y tế, sản xuất, nâng cao hiệu quả nghiên cứu khoa học.
\item Tiêu cực: Có thể gây ô nhiễm môi trường sống, hủy hoại hệ sinh thái nếu không được sử dụng đúng phương pháp và đúng mục đích.
\end{itemize}
\end{kienthuc}

\begin{emdahoc}
\begin{itemize}
\item Đối tượng nghiên cứu chủ yếu của Vật lí là các dạng vận động của vật chất và năng lượng.
\item Phương pháp nghiên cứu Vật lí thường sử dụng là phương pháp thực nghiệm và phương pháp lí thuyết.
\item Vật lí được coi là cơ sở của khoa học tự nhiên và công nghệ, có ảnh hưởng mạnh mẽ đến sự phát triển của xã hội loài người.
\item Việc ứng dụng các thành tựu vật lí cần được kiểm soát đúng phương pháp để tránh gây ô nhiễm môi trường và hủy hoại hệ sinh thái.
\end{itemize}
\end{emdahoc}
